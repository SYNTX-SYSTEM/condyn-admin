def g(W, *args): return subprocess.run(["git", "-c", "commit.gpgsign=false", *args], cwd=W, env=ENV, capture_output=True, text=True, check=True).stdout
def write_new(W, rel, text):
    path = f"{W}/{rel}"; os.makedirs(os.path.dirname(path), exist_ok=True)
    tmp = f"{path}.pink-control"
    with open(tmp, "w") as f: f.write(text)

def read(W, rel): return open(f"{W}/{rel}").read()

def commit(W, msg): g(W, "add", "-A"); g(W, "commit", "-q", "-m", msg)
def main_at(W, rev): g(W, "checkout", "-q", "-B", "pinkmain", rev)
def set_main(W): g(W, "update-ref", "refs/remotes/origin/main", "HEAD")
