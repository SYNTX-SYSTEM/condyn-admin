const decode = (s: string) => s.replace(/<!-- -->/g, "").replace(/<[^>]+>/g, "").replace(/&amp;/g, "&").replace(/&lt;/g, "<").replace(/&gt;/g, ">").replace(/&quot;/g, '"').replace(/&#x27;/g, "'");
/** The element carrying `attribute` inside `html`, balanced by its tag name. */
function element(html: string, attribute: string): string | null {
  const at = html.indexOf(attribute);
  if (at < 0) return null;
  const start = html.lastIndexOf("<", at);
  const tag = /^<(\w+)/.exec(html.slice(start))![1];
  let depth = 0;
  const re = new RegExp(`<(/?)${tag}\\b[^>]*>`, "g");
  re.lastIndex = start;
  for (let m = re.exec(html); m; m = re.exec(html)) { depth += m[1] ? -1 : 1; if (depth === 0) return html.slice(start, m.index + m[0].length); }
  return null;
}

describe("PINK: role cards state only what the pool and the matching state (GRÜN frontend/role-cards)", () => {
  it("S-1 … S-6 at the organization level", () => {
    const html = renderToString(<SemanticCareerIntelligenceField data={EMPTY_CAREER_INTELLIGENCE_DATA} fieldFocus="FIELD" jobPoolUploadId="JPOOL_RC" analysisId="ANL_TEST" jobOrganizationId="ORG_1" />);
    expect(html).toContain('data-observation-kind="ORGANIZATION"');
    const report: unknown[] = [];
    for (const role of ROLES) {
      const card = element(html, `data-testid="syntx-role-${role.id}"`);
      expect(card, role.id).not.toBeNull();
      const box = /data-card-box="(\w+)"/.exec(card!)?.[1];
      const parts = role.seniority.split("/").map(p => p.trim()).filter(Boolean);
      const standing = decode(element(card!, "data-sx-role-standing")!);
      const title = decode(element(card!, "data-sx-role-title")!);
      const domainEl = element(card!, "data-sx-role-domain");
      const requirements = decode(element(card!, "data-sx-role-requirements")!);
      const matchingEl = element(card!, "data-sx-role-matching")!;
      const matching = decode(matchingEl);
      const dot = /<span[^>]*style="([^"]*)"/.exec(matchingEl.slice(matchingEl.indexOf(">") + 1))![1];
      report.push({ id: role.id, box, standing, title, domain: domainEl && decode(domainEl), requirements, matching, dot });
      expect(standing, `S-1 ${role.id}`).toBe(parts.length ? parts.map(p => p.toUpperCase()).join(" · ") : "ROLE");
      expect(title.replace(/ /g, " "), `S-2 ${role.id}`).toBe(role.title);
      expect(title, `S-2 ${role.id} dash bound`).toBe(role.title.replace(/ ([–—-]) /g, " $1 "));
      expect(["orbit", "packed"], `S-3 ${role.id} box`).toContain(box);
      if (box === "orbit" && role.domain !== null) expect(domainEl && decode(domainEl), `S-3 ${role.id}`).toBe(role.domain); else expect(domainEl, `S-3 ${role.id}`).toBeNull();
      expect(requirements, `S-4 ${role.id}`).toBe(`${role.requirements} ${role.requirements === 1 ? "REQUIREMENT" : "REQUIREMENTS"}`);
      expect(card!, `S-4 ${role.id}`).toContain(`data-requirement-count="${role.requirements}"`);
      expect(matching, `S-5 ${role.id}`).toBe(role.matched ? "MATCHING READ" : "NO MATCHING READ");
      expect(card!, `S-5 ${role.id}`).toContain(`data-matching-read="${role.matched}"`);
      expect(/background:transparent/.test(dot), `S-5 ${role.id} mark`).toBe(!role.matched);
      expect(decode(card!), `S-6 ${role.id}`).not.toMatch(/\/\/|resonan|score|rank|%|0\.\d/i);
    }
    console.log(JSON.stringify(report));
  });
});
