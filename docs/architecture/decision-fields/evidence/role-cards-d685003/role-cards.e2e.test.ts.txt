    const overDisc = new Set<string>();
    const fontCuts: Array<{ tag: string; cut: unknown[] }> = [];
    const contrastByTag: Record<string, { ratio: number }> = {};
    let lampCropped = false;
    const titleCrops = new Set<string>();
    const totals = { cards: 0, clampedBeyondFour: 0, needMoreThanCounted: [] as unknown[], ellipsized: [] as unknown[], leastRoomPadding: Infinity, leastRoomContent: Infinity, leastRoomAt: "" };
    for (const run of RUNS) {
      const v = runTag(run);
      const t = (SIL_COPY as any)[run.lang].syntx;
      const page = await open(run);
      for (const org of ORGS) {
        const tag = `${v} ${org.id}`;
        await enter(page, org.id);
        const f = await read(page);
        const items = org.titles.map((title, i) => ({ id: roleId(org.id, i), title }));
        const { layout, compact, compactness } = (layoutLib as any).syntxCosmosLayout(items, f.svgW, f.svgH);
        const drawnIn = compactness.x < 1 || compactness.y < 1;
        const expectedBox = compact ? "compact" : drawnIn ? "orbit" : "packed";
        check("C-1", `${tag}: the in-process factors are the drawn ones`, Number(f.kx) === compactness.x && Number(f.ky) === compactness.y, { dom: [f.kx, f.ky], inProcess: compactness });
        const bad: Record<string, unknown[]> = { box: [], width: [], height: [], lines: [], cut: [], clamp: [], centre: [], texts: [], mark: [], anchor: [], footer: [], fit: [] };
        for (const card of f.cards) {
          totals.cards += 1;
          const role = ROLES.get(card.id.replace("syntx-role-", ""))!;
          const node = layout.nodes.find((n: { id: string }) => n.id === role.id);
          const w = card.r - card.l, h = card.b - card.t, x = (card.l + card.r) / 2, y = (card.t + card.b) / 2;
          // ---- C-1 ----
          if (card.box !== expectedBox) bad.box.push([role.id, card.box]);
          if (Math.abs(w - node.width) > 0.5) bad.width.push([role.id, w, node.width]);
          const wantH = expectedBox === "orbit" ? node.height + ALLOWANCE : node.height;
          if (expectedBox !== "compact" && Math.abs(h - wantH) > 0.5) bad.height.push([role.id, Number(h.toFixed(2)), wantH]);
          const counted = Math.min(4, Math.max(1, Math.floor((node.height - 51) / 18)));
          if (expectedBox !== "compact" && card.titleLines !== String(counted)) bad.lines.push([role.id, card.titleLines, counted]);
          // ---- C-2 ----
          const padTop = card.t + card.border, padBottom = card.b - card.border;
          for (const line of card.lines) {
            const roomPad = Math.min(line.t - padTop, padBottom - line.b), roomContent = Math.min(line.t - padTop - card.padTop, padBottom - card.padBottom - line.b);
            if (roomPad < totals.leastRoomPadding) { totals.leastRoomPadding = roomPad; totals.leastRoomAt = `${tag} ${role.id} ${line.kind}`; }
            totals.leastRoomContent = Math.min(totals.leastRoomContent, roomContent);
            if (roomPad < -0.5 || line.l < card.l + card.border - 0.5 || line.r > card.r - card.border + 0.5) bad.cut.push([role.id, line.kind, Number(roomPad.toFixed(2))]);
            if (line.kind !== "title" && line.scrollW > line.clientW + 1) totals.ellipsized.push({ tag, id: role.id, kind: line.kind, text: line.text, cardWidth: Number(w.toFixed(1)), scrollW: line.scrollW, clientW: line.clientW });
          }
          const clamped = card.titleScrollH > card.titleClientH + 1;
          if (expectedBox !== "compact" && card.titleNaturalBase > counted) totals.needMoreThanCounted.push({ tag, id: role.id, title: role.title, counted, naturalAtBase: card.titleNaturalBase, size: card.titleSize, cut: clamped, width: node.width, box: card.box });
          if (expectedBox !== "compact" && clamped && card.titleNaturalBase <= 4) bad.clamp.push([role.id, role.title, { counted, naturalAtBase: card.titleNaturalBase, size: card.titleSize }]);
          if (clamped && card.titleNaturalBase > 4) totals.clampedBeyondFour += 1;
          if (card.titleSize !== null) titleSizes.push(card.titleSize);
          // ---- C-9 (4ecfe67): the title read smaller in half pixels down to 10 px, "…" only below that; full texts as tooltips ----
          if (FIT_TIP) {
            const size = Number(card.titleSize);
            if (card.titleSize === null || !(size >= 10 && size <= (expectedBox === "compact" ? 11 : 11.5)) || (clamped && size > 10)) bad.fit.push([role.id, { size: card.titleSize, cut: clamped }]);
            if (card.titleTip !== role.title) bad.fit.push([role.id, "title tooltip", card.titleTip]);
            const standingLine = card.lines.find(l => l.kind === "standing"), domainLine = card.lines.find(l => l.kind === "domain");
            if (standingLine && (standingLine.tip ?? "") !== role.seniority) bad.fit.push([role.id, "standing tooltip", standingLine.tip, role.seniority]);
            if (domainLine && domainLine.tip !== role.domain) bad.fit.push([role.id, "domain tooltip", domainLine.tip]);
          }
          // ---- C-3 ----
          for (const line of card.lines) if (Math.abs((line.l + line.r) / 2 - x) > 1) bad.centre.push([role.id, line.kind, Number(((line.l + line.r) / 2 - x).toFixed(2))]);
          for (const row of card.titleRows) if (Math.abs((row.l + row.r) / 2 - x) > 1) bad.centre.push([role.id, "title row", Number(((row.l + row.r) / 2 - x).toFixed(2))]);
          // ---- C-4 ----
          const parts = role.seniority.split("/").map(p => p.trim()).filter(Boolean);
          const text = (kind: Kind) => card.lines.find(l => l.kind === kind)?.text ?? null;
          const want = {
            standing: parts.length ? parts.map(p => p.toUpperCase()).join(" · ") : t.manifest.role,
            title: role.title.replace(/ ([–—-]) /g, " $1 "),
            domain: expectedBox === "orbit" && role.domain.length > 0 ? role.domain : null,
            requirements: `${role.requirements} ${role.requirements === 1 ? t.manifest.requirement : t.requirements}`,
            matching: t.manifest.matchingNotRead
          };
          for (const kind of KINDS) if (text(kind) !== want[kind]) bad.texts.push([role.id, kind, text(kind), want[kind]]);
          if (card.reqAttr !== String(role.requirements) || card.matchAttr !== "false") bad.texts.push([role.id, "attributes", card.reqAttr, card.matchAttr]);
          if (/\/\/|\bresonance\b|\bscore\b|\brank\b|%/i.test(card.cardText.replace(role.title, ""))) bad.texts.push([role.id, "forbidden", card.cardText]);
          if (card.markBackground === null || rgba(card.markBackground).a !== 0) bad.mark.push([role.id, card.markBackground]);
          // ---- C-8 footer: both lines whole, at one size, at most 8.5 px ----
          const req = card.lines.find(l => l.kind === "requirements"), mat = card.lines.find(l => l.kind === "matching");
          if (req && mat) {
            footerSizes.push(`${run.lang}${expectedBox === "compact" ? " compact" : ""} ${req.fontSize}`);
            if (req.scrollW > req.clientW + 1 || mat.scrollW > mat.clientW + 1 || Math.abs(req.fontSize - mat.fontSize) > 0.01 || req.fontSize > 8.5) bad.footer.push([role.id, { req: [req.text, req.scrollW, req.clientW, req.fontSize], mat: [mat.text, mat.scrollW, mat.clientW, mat.fontSize], width: Number(w.toFixed(1)) }]);
            if (run.lang === "en" && !run.font && req.fontSize !== 8.5) bad.footer.push([role.id, "English below 8.5 px", req.fontSize]);
          }
          // ---- C-6 ----
          if (card.anchor) {
            const dx = f.cx - x, dy = f.cy - y, len = Math.hypot(dx, dy);
            const ax = card.anchor.x - x, ay = card.anchor.y - y;
            const edge = Math.min(Math.abs(Math.abs(ax) - w / 2), Math.abs(Math.abs(ay) - h / 2));
            const inside = Math.abs(ax) <= w / 2 + 1 && Math.abs(ay) <= h / 2 + 1;
            const off = Math.abs(ax * dy - ay * dx) / len;
            if (!(edge <= 1.5 && inside && off <= 1.5 && ax * dx + ay * dy > 0)) bad.anchor.push([role.id, { edge: Number(edge.toFixed(2)), off: Number(off.toFixed(2)) }]);
          } else bad.anchor.push([role.id, "no anchor"]);
        }
        measured[tag] = { factors: [f.kx, f.ky], box: expectedBox, svg: [f.svgW, f.svgH], cards: f.cards.map(c => [c.id.replace("syntx-role-", ""), Number((c.r - c.l).toFixed(1)), Number((c.b - c.t).toFixed(1)), c.titleLines]) };
        check("C-1", `${tag}: data-card-box "${expectedBox}" on every card (factors ${f.kx}/${f.ky})`, bad.box.length === 0 && f.cards.length === items.length, bad.box);
        check("C-1", `${tag}: rendered width = the estimate's, height = ${expectedBox === "orbit" ? "estimate + 12" : expectedBox === "packed" ? "the estimate" : "(compact form)"} (±0.5 px), title lines = the counted ones`, bad.width.length + bad.height.length + bad.lines.length === 0, { width: bad.width, height: bad.height, lines: bad.lines });
        check("C-2", `${tag}: nothing cut (every line inside the padding box), no title clamped that fits in four lines`, bad.cut.length + bad.clamp.length === 0, { cut: bad.cut, clamp: bad.clamp });
        check("C-3", `${tag}: every line and title row centred on the card's axis (±1 px)`, bad.centre.length === 0, bad.centre);
        check("C-4", `${tag}: standing, title, domain, requirements, matching as the pool states them; hollow mark; nothing else`, bad.texts.length + bad.mark.length === 0, { texts: bad.texts, mark: bad.mark });
        if (FOOTER_TIP && !run.font) check("C-8", `${tag}: footer (requirements, matching) whole on every card, both at one size ≤ 8.5 px${run.lang === "en" && !run.font ? ", 8.5 px in English" : ""}`, bad.footer.length === 0, bad.footer);
        else measured[`${tag} footer cut`] = bad.footer.filter(b => typeof (b as unknown[])[1] === "object");
        if (FOOTER_TIP && run.font) fontCuts.push({ tag, cut: bad.footer });
        if (FIT_TIP) check("C-9", `${tag}: titles at 10 … ${"11.5"} px in the counted lines, "…" only where 10 px does not fit; the full title, seniority and domain as tooltips`, bad.fit.length === 0, bad.fit);
        check("C-6", `${tag}: the tether lamp on the card's edge where the line from the centre meets it`, bad.anchor.length === 0, bad.anchor);
        // ---- C-5 contrast ----
        if (READ_CONTRAST(run)) {
          const surface = await page.evaluate(() => { const r = document.querySelector('[data-sx-surface="header"]')?.getBoundingClientRect(); return r ? { l: r.left, t: r.top, r: r.right, b: r.bottom } : null; });
          // ---- C-7 the tether lamp and its conduction stub, drawn vs. with the card's overflow let visible ----
          const drawn = await sharp(await page.screenshot({ type: "png" })).raw().toBuffer({ resolveWithObject: true });
          const free = await page.addStyleTag({ content: "[data-sx-role-card] { overflow: visible !important; }" });
          await delay(150);
          const freed = await sharp(await page.screenshot({ type: "png" })).raw().toBuffer({ resolveWithObject: true });
          await free.evaluate((node: Element) => node.remove());
          const clipped: unknown[] = [];
          for (const card of f.cards) {
            if (!card.anchor) continue;
            let differing = 0;
            for (let py = Math.max(0, Math.floor(card.anchor.y - 24)); py < Math.min(drawn.info.height, Math.ceil(card.anchor.y + 24)); py += 1) for (let pxl = Math.max(0, Math.floor(card.anchor.x - 24)); pxl < Math.min(drawn.info.width, Math.ceil(card.anchor.x + 24)); pxl += 1) {
              const o = (py * drawn.info.width + pxl) * drawn.info.channels;
              if (Math.max(Math.abs(drawn.data[o] - freed.data[o]), Math.abs(drawn.data[o + 1] - freed.data[o + 1]), Math.abs(drawn.data[o + 2] - freed.data[o + 2])) > 24) differing += 1;
            }
            if (differing >= 10) clipped.push([card.id.replace("syntx-role-", ""), card.box, differing]);
          }
          lampClip.push({ tag, cards: f.cards.length, clipped: clipped.length });
          if (v === "1600x1000" && clipped.length > 0 && !lampCropped) {
            lampCropped = true;
            const c = f.cards.find(x => x.id.endsWith(String((clipped[0] as string[])[0])))!;
            const region = { left: Math.max(0, Math.round(Math.min(c.l, c.anchor!.x) - 30)), top: Math.max(0, Math.round(Math.min(c.t, c.anchor!.y) - 30)) };
            const size = { width: Math.round(Math.max(c.r, c.anchor!.x) + 30) - region.left, height: Math.round(Math.max(c.b, c.anchor!.y) + 30) - region.top };
            for (const [name, img] of [["drawn", drawn], ["overflow-visible", freed]] as const) await sharp(img.data, { raw: { width: img.info.width, height: img.info.height, channels: img.info.channels } }).extract({ ...region, ...size }).resize({ width: size.width * 2, kernel: "nearest" }).png().toFile(resolve(EVIDENCE_DIR, `lamp-${name}.png`));
          }
          check("C-7", `${tag}: the tether lamp and its conduction stub are drawn as on e6493a7 (not clipped by the card's overflow)`, clipped.length === 0, clipped);
          // ---- titles clamped: a crop of each card for the record ----
          for (const card of f.cards) {
            const r = ROLES.get(card.id.replace("syntx-role-", ""))!;
            if (card.titleScrollH > card.titleClientH + 1 && card.titleNatural <= 4 && !titleCrops.has(r.title)) {
              titleCrops.add(r.title);
              await sharp(drawn.data, { raw: { width: drawn.info.width, height: drawn.info.height, channels: drawn.info.channels } }).extract({ left: Math.max(0, Math.round(card.l - 6)), top: Math.max(0, Math.round(card.t - 6)), width: Math.round(card.r - card.l + 12), height: Math.round(card.b - card.t + 12) }).resize({ width: Math.round(card.r - card.l + 12) * 2, kernel: "nearest" }).png().toFile(resolve(EVIDENCE_DIR, `title-clamped-${titleCrops.size}-${v.replace(/[^0-9a-z]+/gi, "-")}.png`));
            }
          }
          // text hidden for the reading; the dev server's own indicator (nextjs-portal) is no part of the field
          const hide = await page.addStyleTag({ content: "[data-sx-role-card] * { color: transparent !important; text-shadow: none !important; } nextjs-portal { display: none !important; }" });
          const disc = await page.evaluate(() => { const r = document.querySelector('[data-testid="syntx-cosmos-centre"]')?.getBoundingClientRect(); return r ? { l: r.left - 6, t: r.top - 6, r: r.right + 6, b: r.bottom + 6 } : null; });
          await delay(150);
          const shot = await page.screenshot({ type: "png" });
          const { data, info } = await sharp(shot).raw().toBuffer({ resolveWithObject: true });
          for (const card of f.cards) for (const line of card.lines) {
            let worst = -1, bg = { r: 0, g: 0, b: 0 }, at = { x: 0, y: 0 };
            const g = line.glyphs;
            // only the card's own glass: pixels under another card (or its lamp, +20 px) or under the orientation surface are left out
            const others = f.cards.filter(o => o.id !== card.id).map(o => ({ l: o.l - 20, t: o.t - 20, r: o.r + 20, b: o.b + 20 })).concat(surface ? [surface] : [], disc ? [disc] : []);
            if (disc && card.l < disc.r && card.r > disc.l && card.t < disc.b && card.b > disc.t) overDisc.add(`${tag} ${card.id.replace("syntx-role-", "")}`);
            for (let py = Math.max(0, Math.floor(g.t)); py < Math.min(info.height, Math.ceil(g.b)); py += 1) for (let pxl = Math.max(0, Math.floor(g.l)); pxl < Math.min(info.width, Math.ceil(g.r)); pxl += 1) {
              if (others.some(o => pxl >= o.l && pxl <= o.r && py >= o.t && py <= o.b)) continue;
              const o = (py * info.width + pxl) * info.channels; const L = lum(data[o], data[o + 1], data[o + 2]);
              if (L > worst) { worst = L; bg = { r: data[o], g: data[o + 1], b: data[o + 2] }; at = { x: pxl - Math.round(card.l), y: py - Math.round(card.t) }; }
            }
            const c = rgba(line.color); const fg = { r: c.r * c.a + bg.r * (1 - c.a), g: c.g * c.a + bg.g * (1 - c.a), b: c.b * c.a + bg.b * (1 - c.a) };
            const L1 = lum(fg.r, fg.g, fg.b), L2 = lum(bg.r, bg.g, bg.b);
            const ratio = (Math.max(L1, L2) + 0.05) / (Math.min(L1, L2) + 0.05);
            if (worst < 0) continue;
            const entry: Record<string, unknown> & { ratio: number } = { ratio: Number(ratio.toFixed(2)), at: `${tag} ${card.id}`, glass: [bg.r, bg.g, bg.b], pixel: at, text: line.color };
            if (ratio < WCAG) entry.topmost = await page.evaluate(([x, y, id]: [number, number, string]) => { const hit = document.elementFromPoint(x, y) as HTMLElement | null; if (!hit) return null; const card = hit.closest("[data-sx-role-card]"); return { tag: hit.tagName.toLowerCase(), testid: hit.closest("[data-testid]")?.getAttribute("data-testid") ?? null, ownCard: card?.getAttribute("data-testid") === id, style: (hit.getAttribute("style") ?? "").slice(0, 120) }; }, [Math.round(card.l) + at.x, Math.round(card.t) + at.y, card.id] as [number, number, string]);
            if (!leastContrast[line.kind] || ratio < leastContrast[line.kind].ratio) leastContrast[line.kind] = entry;
            const key = `${tag} ${line.kind}`;
            if (!(contrastByTag[key]?.ratio <= ratio)) contrastByTag[key] = entry;
          }
          await hide.evaluate((node: Element) => node.remove());
        }
        await page.keyboard.press("Escape");
        await page.locator('[data-testid="syntx-field"][data-observation-kind="FIELD"]').waitFor({ timeout: 30_000 }).catch(() => undefined);
        if ((await page.getByTestId("syntx-field").getAttribute("data-observation-kind")) !== "FIELD") { await page.goto(`${base()}/career/demo?focus=FIELD&jobPoolUploadId=${uploadId}`, { waitUntil: "domcontentloaded" }); await page.getByTestId("field-focus-strip").waitFor({ timeout: 120_000 }); }
      }
      await page.close();
    }
    measured.contrast = leastContrast;
    measured.lampClip = lampClip;
    measured.contrastByTag = contrastByTag;
    measured.titlesClamped = [...titleCrops];
    measured.cardsOverTheCentreDisc = [...overDisc];
    measured.footerInOtherFonts = Object.fromEntries(["DejaVu Sans Mono", "FreeMono", "Fira Code", "IBM Plex Mono"].map(font => [font, fontCuts.filter(c => c.tag.includes(`[${font}]`)).flatMap(c => c.cut.map(x => [c.tag.split(" ").pop(), x]))]));
    measured.titleSizes = Object.fromEntries([...new Set(titleSizes)].sort().map(k => [k, titleSizes.filter(x => x === k).length]));
    measured.footerSizes = Object.fromEntries([...new Set(footerSizes)].sort().map(k => [k, footerSizes.filter(x => x === k).length]));
    measured.totals = { ...totals, leastRoomPadding: Number(totals.leastRoomPadding.toFixed(2)), leastRoomContent: Number(totals.leastRoomContent.toFixed(2)) };
    for (const kind of KINDS) if (leastContrast[kind]) check("C-5", `least contrast of the ${kind} line over every card (1600 × 1000, 1280 × 720) ≥ ${WCAG} : 1`, leastContrast[kind].ratio >= WCAG, leastContrast[kind]);
    check("C-0", "no page errors, no non-GET request from the page", pageErrors.length === 0 && nonGet.length === 0, { pageErrors, nonGet });
    const failed = checks.filter(entry => !entry.ok);
    expect(failed, JSON.stringify(failed, null, 1).slice(0, 20_000)).toEqual([]);
  }, 5_400_000);
});
