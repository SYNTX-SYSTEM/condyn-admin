const scan = (p) => p.evaluate(() => {
  const line = document.querySelector('[data-sx-shell="gate-reason"]'); if (!line) return { line: null };
  const vis = e => { const r = e.getBoundingClientRect(); const cs = getComputedStyle(e); return r.width > 0 && r.height > 0 && cs.visibility !== "hidden" && cs.display !== "none" && +cs.opacity > 0.05; };
  if (!vis(line)) return { line: { visible: false } };
  const L = line.getBoundingClientRect(); const VA = innerWidth * innerHeight;
  const inter = r => Math.max(0, Math.min(r.right, L.right) - Math.max(r.left, L.left)) * Math.max(0, Math.min(r.bottom, L.bottom) - Math.max(r.top, L.top));
  const covered = [];
  for (const e of document.querySelectorAll("body *")) {
    if (e === line || line.contains(e) || e.contains(line) || !vis(e)) continue;
    const r = e.getBoundingClientRect(); const a = inter(r); if (a <= 0) continue;
    if (r.width * r.height > 0.4 * VA) continue; // stage / background containers
    const interactive = e.matches('button, a[href], input, select, textarea, [role="button"], [tabindex="0"]');
    const text = [...e.childNodes].some(n => n.nodeType === 3 && n.textContent.trim());
    const tid = e.dataset?.testid ?? null;
    if (!interactive && !text && !tid) continue;
    covered.push({ tid, tag: e.tagName.toLowerCase(), interactive, text: text ? e.textContent.trim().slice(0, 40) : null, px: Math.round(a), share: +(a / (r.width * r.height)).toFixed(3), rect: [r.left, r.top, r.width, r.height].map(Math.round) });
  }
  const intake = document.querySelector('[data-sx-panel="intake"]'); const I = intake && vis(intake) ? intake.getBoundingClientRect() : null;
  const act = line.querySelector('[data-testid="field-entry-action"]')?.getBoundingClientRect();
  const sel = document.querySelector('[data-sx-shell="selector"]')?.getBoundingClientRect(); const inv = document.querySelector('[data-sx-shell="investigation"]')?.getBoundingClientRect();
  const box = document.querySelector('[data-testid="syntx-pool"]')?.getBoundingClientRect();
  const org = document.querySelector('[data-testid="syntx-organization-org_kestrel"]')?.getBoundingClientRect();
  return { line: { visible: true, entry: line.dataset.entryState, rect: [L.left, L.top, L.width, L.height].map(Math.round), position: getComputedStyle(line).position, overflowX: line.scrollWidth > line.clientWidth + 1, overflowY: line.scrollHeight > line.clientHeight + 1, offscreen: L.right > innerWidth + 0.5 || L.left < -0.5 },
    action: act ? [act.left, act.top, act.width, act.height].map(Math.round) : null, covered,
    intake: I ? { rect: [I.left, I.top, I.width, I.height].map(Math.round), gapBelowLine: Math.round(I.top - L.bottom), overlapsLine: inter(I) > 0 } : null,
    selector: sel ? Math.round(sel.bottom) : null, investigation: inv ? [inv.left, inv.top, inv.width, inv.height].map(Math.round) : null,

    orgRing: org ? [org.left, org.top, org.width, org.height].map(Math.round) : null, pageScrollX: document.documentElement.scrollWidth > innerWidth + 1 };
});
