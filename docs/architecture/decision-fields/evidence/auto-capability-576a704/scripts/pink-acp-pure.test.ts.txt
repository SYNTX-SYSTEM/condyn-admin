// ---- D: an independent PINK oracle of ROLE_RANKING_V1 from the spec text and GELB's Q3, compared with the implementation ----
type Req = { id: string; necessity: string; basis: string; finding: SemanticFinding | null };
type Role = { pool: string; id: string; title: string; reqs: Req[] };
const F = (state: SemanticFinding["state"], name = "cap"): SemanticFinding => ({ state, capabilityName: name, explanation: `e-${state}`, evidenceLimitation: "l", quotes: [{ sourceDocumentId: "CV", exactQuote: "q" }] });
function oracle(roles: Role[]) {
  const W: Record<string, number> = { REQUIRED: 3, PREFERRED: 2, OPTIONAL: 1 }; const SV: Record<string, number> = { EQUIVALENT: 1, COVERS: 1, PARTIAL: 0.5, DIFFERENT: 0, UNCLEAR: 0 };
  const rows = roles.map(r => { const q = r.reqs.map(x => { const w = W[x.necessity] ?? 1; const v = x.finding ? SV[x.finding.state] : (x.basis === "NAME_EXACT" || x.basis === "NAME_ALIAS" ? 0.5 : 0); return { w, v, req: x.necessity === "REQUIRED", sem: !!x.finding }; });
    const sw = q.reduce((n, x) => n + x.w, 0); return { key: `${r.pool}/${r.id}`, title: r.title, pool: r.pool, id: r.id, tier: q.some(x => x.sem) ? 0 : 1, fit: sw > 0 ? Math.round((q.reduce((n, x) => n + x.w * x.v, 0) / sw) * 1e6) / 1e6 : null, gaps: q.filter(x => x.req && x.v === 0).length }; });
  const c = (a: string, b: string) => (a < b ? -1 : a > b ? 1 : 0);
  return rows.sort((a, b) => a.tier - b.tier || (a.fit === null ? 1 : 0) - (b.fit === null ? 1 : 0) || (b.fit ?? 0) - (a.fit ?? 0) || a.gaps - b.gaps || c(a.title, b.title) || c(a.pool, b.pool) || c(a.id, b.id)).map(r => `${r.key} ${r.tier ? "LEXICAL_ONLY" : "SEMANTIC"} ${r.fit} g${r.gaps}`);
}
function discoveryOf(roles: Role[]) {
  const findings = new Map<string, SemanticFinding | null>();
  const discovery = { pools: [...new Set(roles.map(r => r.pool))].map(p => ({ jobPoolUploadId: p, poolName: `name-${p}` })), roles: roles.map(r => ({ key: { jobPoolUploadId: r.pool, poolRoleId: r.id }, title: r.title, organizationName: "o",
    requirements: r.reqs.map(x => { const key = { jobPoolUploadId: r.pool, poolRoleId: r.id, poolRequirementId: x.id }; findings.set(`${r.pool}/${r.id}/${x.id}`, x.finding);
      return { key, requirementText: x.id, necessity: x.necessity, lexical: x.basis === "NONE" ? { basis: "NONE", proposalId: null, proposalName: null, matchedConstituent: null, evidence: [], quoteNamesRequirement: null } : { basis: x.basis, proposalId: "p", proposalName: "lex", matchedConstituent: null, evidence: [{ sourceDocumentId: "CV", exactQuote: "lq" }], quoteNamesRequirement: true } }; }) })) };
  return { discovery: discovery as never, findingFor: (q: any) => findings.get(`${q.key.jobPoolUploadId}/${q.key.poolRoleId}/${q.key.poolRequirementId}`) ?? null };
}
