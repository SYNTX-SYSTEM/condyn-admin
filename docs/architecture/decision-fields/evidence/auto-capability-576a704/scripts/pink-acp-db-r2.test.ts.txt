/** Whole-database fingerprint: every table of the schema with its column list and an md5 over all rows. */
async function fingerprint() {
  const tables = (await q<{ t: string }>(`SELECT tablename AS t FROM pg_tables WHERE schemaname = current_schema() ORDER BY 1`)).map(r => r.t);
  const out: Record<string, string> = {};
  for (const t of tables) {
    const cols = (await q<{ c: string }>(`SELECT string_agg(column_name || ':' || data_type, ',' ORDER BY ordinal_position) AS c FROM information_schema.columns WHERE table_schema = current_schema() AND table_name = '${t}'`))[0].c;
    const h = (await q<{ h: string | null; n: number }>(`SELECT md5(coalesce(string_agg(x::text, '|' ORDER BY x::text), '')) AS h, count(*)::int AS n FROM "${t}" x`))[0];
    out[t] = `${h.n} ${h.h} ${createHashShort(cols)}`;
  }
  return out;
}
const createHashShort = (s: string) => createHash("sha256").update(s ?? "").digest("hex").slice(0, 12);
const diff = (a: Record<string, string>, b: Record<string, string>) => [...new Set([...Object.keys(a), ...Object.keys(b)])].filter(k => a[k] !== b[k]).map(k => `${k}: ${a[k] ?? "absent"} -> ${b[k] ?? "absent"}`);
