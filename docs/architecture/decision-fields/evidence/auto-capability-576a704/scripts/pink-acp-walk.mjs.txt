async function ctxFor(w, h, dsf) {
  const ctx = await browser.newContext({ viewport: { width: w, height: h }, deviceScaleFactor: dsf });
  await ctx.addInitScript(() => { try { localStorage.setItem("condyn_onboarding_seen", "true"); } catch {} });
  const page = await ctx.newPage(); const requests = [], errors = [];
  page.on("request", r => { const u = new URL(r.url()); if (u.pathname.startsWith("/api/")) requests.push({ at: Date.now(), m: r.method(), p: u.pathname }); });
  page.on("pageerror", e => errors.push(String(e).slice(0, 160)));
  return { ctx, page, requests, errors };
}
