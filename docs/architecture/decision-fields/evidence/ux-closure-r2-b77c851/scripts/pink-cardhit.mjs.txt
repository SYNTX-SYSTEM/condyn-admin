// PINK §10 V-19: at the organisation level, can each visible role card be clicked by the mouse (trial click = actionability incl. the
// hit target, no action), per viewport and mode? GET only in the browser. argv: <base> <out dir> [real|standin] [repo root]
// standin: the DLL pool is uploaded first, outside the browser (PINK's own lab only).
import { readFileSync, writeFileSync } from "node:fs";
import { pathToFileURL } from "node:url";
const [BASE, OUT, KIND = "real", ROOT = ""] = process.argv.slice(2);
if (KIND === "standin") await fetch(`${BASE}/api/career/job-pools`, { method: "POST", headers: { "content-type": "application/json", "x-condyn-principal-actor-id": "JOB_POOL_UPLOADER:PINK_S10" }, body: readFileSync(`${ROOT}/docs/examples/job-pool-tests/DLL_JOBPOOL_ENGINEERING.json`, "utf8") });
const pw = await import(pathToFileURL(process.env.CONDYN_PLAYWRIGHT_MODULE).href); const { chromium } = pw.chromium ? pw : pw.default;
const b = await chromium.launch(); let blocked = 0; const res = [];
const DLL = "JPOOL_B94850C439974E27496FE3B1A9A076B5";
for (const [n, w, h, d] of [["1920x1080", 1920, 1080, 1], ["1600x900", 1600, 900, 1], ["1440x900", 1440, 900, 1], ["1280x720", 1280, 720, 1], ["zoom200-1920x1080", 960, 540, 2], ["zoom200-1280x720", 640, 360, 2]]) {
  const ctx = await b.newContext({ viewport: { width: w, height: h }, deviceScaleFactor: d }); await ctx.addInitScript(() => { try { localStorage.setItem("condyn_onboarding_seen", "true"); } catch {} });
  await ctx.route("**/*", r => { const m = r.request().method(); if (m !== "GET" && m !== "HEAD") { blocked += 1; return r.abort(); } return r.continue(); });
  const p = await ctx.newPage();
  for (const [scene, mode] of [["org-PXJ", "PERSON_X_JOB"], ["org-PERSON", "PERSON"]]) {
    await p.goto(`${BASE}/career/demo?investigationMode=${mode}&jobPoolUploadId=${DLL}&jobOrganizationId=org_kestrel`, { waitUntil: "domcontentloaded" }); await new Promise(r => setTimeout(r, 7000));
    const out = {};
    for (const role of ["role_kestrel_fullstack", "role_kestrel_frontend_architect", "role_kestrel_engineering_lead"]) {
      const loc = p.getByTestId(`syntx-role-${role}`); if (!(await loc.count())) { out[role] = "absent"; continue; }
      await loc.scrollIntoViewIfNeeded({ timeout: 3000 }).catch(() => {}); await new Promise(r => setTimeout(r, 400));
      const samples = []; for (let k = 0; k < 6; k += 1) { samples.push(await loc.evaluate(e => { const r = e.getBoundingClientRect(); const pts = [[0.5, 0.5], [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75]].map(([fx, fy]) => [r.left + r.width * fx, r.top + r.height * fy]); let self = 0, inView = 0; const cover = new Set(); for (const [x, y] of pts) { if (x < 0 || y < 0 || x >= innerWidth || y >= innerHeight) continue; inView += 1; const c = document.elementFromPoint(x, y); if (c && e.contains(c)) self += 1; else if (c) cover.add(c.closest("[data-testid]")?.dataset.testid ?? c.tagName.toLowerCase()); } return { box: [Math.round(r.left), Math.round(r.top), Math.round(r.width), Math.round(r.height)], self, inView, cover: [...cover] }; })); await new Promise(r => setTimeout(r, 500)); }
      const best = Math.max(...samples.map(x => x.self)), centreSelf = samples.filter(x => x.self > 0).length;
      const geo = { box: samples[0].box, hitSelfBest: `${best}/5`, samplesWithAnyHit: `${centreSelf}/6`, cover: [...new Set(samples.flatMap(x => x.cover))].slice(0, 4) }; const trial = "n/a (orbiting: never stable)";
      out[role.replace("role_kestrel_", "")] = { ...geo, trial };
    }
    res.push({ vp: n, scene, ...out });
    if (n === "1280x720") await p.screenshot({ path: `${OUT}/V19-${KIND}-${scene}-1280x720.jpg`, type: "jpeg", quality: 75 });
  }
  await ctx.close();
}
writeFileSync(`${OUT}/cardhit-${KIND}.json`, JSON.stringify({ kind: KIND, blocked, res }, null, 1));
for (const r of res) console.log(r.vp, r.scene, ["fullstack", "frontend_architect", "engineering_lead"].map(k => `${k}:${r[k]?.hitSelfBest} any${r[k]?.samplesWithAnyHit} cover[${r[k]?.cover}] box${JSON.stringify(r[k]?.box)}`).join("  "));
console.log("DONE blocked", blocked); await b.close();
