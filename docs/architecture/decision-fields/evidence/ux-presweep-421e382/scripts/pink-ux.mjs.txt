// Tab through the page: order, visibility, indicator, URL / history changes without Enter (FOCUS ≠ SELECTION), traps.
const tabSweep = async (page, name, max = 160, shift = false) => {
  await page.evaluate(PAGE_HELPERS); await page.evaluate(() => { document.activeElement?.blur?.(); window.scrollTo(0, 0); });
  const startUrl = page.url(), startH = await page.evaluate(() => history.length); const order = []; let lastKey = null, repeats = 0;
  for (let i = 0; i < max; i += 1) {
    await page.keyboard.press(shift ? "Shift+Tab" : "Tab"); await sleep(120);
    const f = await focusSnap(page); await sleep(220); const u1 = await prevStyle(page); await sleep(260); const u2 = await prevStyle(page);
    if (order.length) { const last = order.at(-1); last.indicatorChanged = indicator(last.style, u1, u2); last.animatedUnfocused = u1 && u2 ? Object.keys(u1).filter(k => u1[k] !== u2[k]) : null; delete last.style; }

  }
  const last = order.at(-1); if (last) delete last.style;
  const changes = order.filter((f, i) => i > 0 && (f.url !== order[i - 1].url || f.h !== order[i - 1].h)).map(f => ({ at: f.testid, url: f.url, h: f.h }));
  const res = { name, shift, steps: order.length, startUrlChangedByTabbing: page.url() !== startUrl, historyDelta: (await page.evaluate(() => history.length)) - startH, urlChangesDuringTab: changes, trapped: repeats >= 3 ? lastKey : null,
    focusOnHidden: order.filter(f => f.box?.hidden).map(f => f.testid ?? f.label), focusOutsideViewport: order.filter(f => f.box && !f.box.hidden && !f.box.inViewport).map(f => f.testid ?? f.label),
    noIndicator: order.filter(f => f.indicatorChanged && f.indicatorChanged.length === 0 && !f.body).map(f => f.testid ?? f.label), unnamed: order.filter(f => !f.body && !f.label && !f.testid).length,
    order: order.map(f => f.body ? "<body>" : `${f.testid ?? f.tag}${f.label ? ` «${f.label.slice(0, 40)}»` : ""}${f.box?.hidden ? " [hidden]" : ""}${f.box && !f.box.inViewport ? " [off-viewport]" : ""}${f.indicatorChanged && f.indicatorChanged.length === 0 ? " [no indicator]" : ""}`) };
  return res;
};
