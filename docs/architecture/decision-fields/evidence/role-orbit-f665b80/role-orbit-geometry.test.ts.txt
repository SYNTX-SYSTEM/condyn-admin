const ringOf = (l: CapabilityCosmosLayout, n: Node) => l.rings.find(r => r.index === n.ringIndex)!;
const at = (l: CapabilityCosmosLayout, n: Node, phase: number) => { const r = ringOf(l, n); return { x: Math.cos(n.angle + phase) * r.radiusX, y: Math.sin(n.angle + phase) * r.radiusY }; };
const air = (p: { x: number; y: number }, a: Node, q: { x: number; y: number }, b: Node, e: number) => Math.max(Math.abs(p.x - q.x) - (a.width + b.width) / 2 - GAP.x, Math.abs(p.y - q.y) - (a.height + b.height + 2 * e) / 2 - GAP.y);

function centreAir(l: CapabilityCosmosLayout, e: number, phases = FINE) { let m = Infinity; for (const n of l.nodes) for (const ph of phases) { const p = at(l, n, ph); m = Math.min(m, Math.hypot(Math.max(0, Math.abs(p.x) - n.width / 2), Math.max(0, Math.abs(p.y) - (n.height + e) / 2)) - CENTRE); } return m; }
function sameOrbit(l: CapabilityCosmosLayout, e: number, phases = FINE) { let m = Infinity, t = Infinity; for (const a of l.nodes) for (const b of l.nodes) if (a.id < b.id && a.ringIndex === b.ringIndex) for (const ph of phases) { const p = at(l, a, ph), q = at(l, b, ph); m = Math.min(m, air(p, a, q, b, e)); t = Math.min(t, touch(p, a, q, b, e)); } return [m, t]; }
/** Across two orbits in motion: per distinct card size on each orbit, every phase of one against every phase of the other. */
function acrossInMotion(l: CapabilityCosmosLayout, ringA: number, ringB: number, e: number, stopBelow = -Infinity) {
  const sizes = (ring: number) => [...new Map(l.nodes.filter(n => n.ringIndex === ring).map(n => [`${n.width}x${n.height}`, n])).values()];
  const A = l.rings.find(r => r.index === ringA)!, B = l.rings.find(r => r.index === ringB)!;
  const ax = CROSS_COS.map(c => c * A.radiusX), ay = CROSS_SIN.map(c => c * A.radiusY), bx = CROSS_COS.map(c => c * B.radiusX), by = CROSS_SIN.map(c => c * B.radiusY);
  let m = Infinity;
  for (const a of sizes(ringA)) for (const b of sizes(ringB)) {
    const W = (a.width + b.width) / 2 + GAP.x, H = (a.height + b.height + 2 * e) / 2 + GAP.y;
    for (let i = 0; i < ax.length; i += 1) { for (let j = 0; j < bx.length; j += 1) { const v = Math.max(Math.abs(ax[i] - bx[j]) - W, Math.abs(ay[i] - by[j]) - H); if (v < m) m = v; } if (m < stopBelow) return m; }
  }
  return m;
}

/** Across every pair of orbits in motion, per pair of card sizes (each card its own size, height + e): the least air over 0.5° x 0.5°. */
function acrossBySize(l: CapabilityCosmosLayout, e: number): Map<string, number> {
  const out = new Map<string, number>();
  const ids = ringIds(l);
  for (let i = 0; i < ids.length; i += 1) for (let j = i + 1; j < ids.length; j += 1) {
    const A = l.rings.find(r => r.index === ids[i])!, B = l.rings.find(r => r.index === ids[j])!;
    const ax = HALF_COS.map(c => c * A.radiusX), ay = HALF_SIN.map(c => c * A.radiusY), bx = HALF_COS.map(c => c * B.radiusX), by = HALF_SIN.map(c => c * B.radiusY);
    const sizes = (ring: number) => [...new Map(l.nodes.filter(n => n.ringIndex === ring).map(n => [`${n.width}x${n.height}`, n])).values()];
    for (const a of sizes(ids[i])) for (const b of sizes(ids[j])) {

      out.set(`rings ${ids[i]}-${ids[j]} ${a.width}x${a.height}/${b.width}x${b.height}`, m);
    }
  }
  return out;
}

interface SizePair { ra: number; rb: number; a: Node; b: Node; floor?: number; fineFloor?: number }

interface PackedState { before: CapabilityCosmosLayout; pairs?: SizePair[]; rest?: number }
const quarterGrid = (n: number, offset: number) => { const ph = Array.from({ length: n }, (_, i) => deg((i * 360) / n + offset)); return { cos: ph.map(Math.cos), sin: ph.map(Math.sin) }; };
const GRID_1 = quarterGrid(360, 0.3), GRID_QUARTER = quarterGrid(1440, 0.07);
/** The least air in motion between a card of size a on ring ra and one of size b on ring rb, every phase pair of the grid. */
function pairAir(l: CapabilityCosmosLayout, sp: SizePair, e: number, grid = GRID_1, stopBelow = -Infinity): number {
  const A = l.rings.find(r => r.index === sp.ra)!, B = l.rings.find(r => r.index === sp.rb)!;
  const W = (sp.a.width + sp.b.width) / 2 + GAP.x, H = (sp.a.height + sp.b.height + 2 * e) / 2 + GAP.y;
  const ax = grid.cos.map(c => c * A.radiusX), ay = grid.sin.map(c => c * A.radiusY), bx = grid.cos.map(c => c * B.radiusX), by = grid.sin.map(c => c * B.radiusY);
  let m = Infinity;
  for (let p = 0; p < ax.length; p += 1) { for (let q = 0; q < bx.length; q += 1) { const v = Math.max(Math.abs(ax[p] - bx[q]) - W, Math.abs(ay[p] - by[q]) - H); if (v < m) m = v; } if (m < stopBelow) return m; }
  return m;
}
/**
 * The rule at (kx, ky) with the rule's card height (the least air; negative: broken), early out once broken: centre and own orbit;
 * across every pair of orbits and every pair of card sizes in motion never below min(0, the packing's own air of that pair); at rest
 * never closer than the packing. A tighter pair that holds on the 1° grid is read again every 0.25° across (fine = true).
 */
function ruleAir(before: CapabilityCosmosLayout, kx: number, ky: number, state: PackedState, phases = FINE, fine = false): number {
  const l = scaled(before, kx, ky);
  let m = Math.min(centreAir(l, RULE_EXCESS, phases), sameOrbit(l, RULE_EXCESS, phases)[0]);
  if (m < HOLDS_MARGIN) return m;
  if (!state.pairs) {
    const ids = ringIds(before); state.pairs = [];
    const sizes = (ring: number) => [...new Map(before.nodes.filter(n => n.ringIndex === ring).map(n => [`${n.width}x${n.height}`, n])).values()];
    for (let i = 0; i < ids.length; i += 1) for (let j = i + 1; j < ids.length; j += 1) for (const a of sizes(ids[i])) for (const b of sizes(ids[j])) state.pairs.push({ ra: ids[i], rb: ids[j], a, b });
    state.rest = acrossAtRest(before, RULE_EXCESS);
  }
  for (const sp of state.pairs) {
    if (fine) sp.fineFloor ??= pairAir(before, sp, RULE_EXCESS, GRID_QUARTER); else sp.floor ??= pairAir(before, sp, RULE_EXCESS, GRID_1);
    const floor = Math.min(0, fine ? sp.fineFloor! : sp.floor!);
    m = Math.min(m, pairAir(l, sp, RULE_EXCESS, fine ? GRID_QUARTER : GRID_1, HOLDS_MARGIN + floor) - floor);
    if (m < HOLDS_MARGIN) return m;
  }
  if (state.pairs.length > 0) m = Math.min(m, acrossAtRest(l, RULE_EXCESS) - Math.min(0, state.rest!));
  return m;
}

      if (before.staticPositions || ringIds(before).length < 2) continue;
      multi += 1;
      const { layout, compactness } = syntxCosmosLayout(items, width, height);
      if (compactness.x === 1 && compactness.y === 1) continue;
      drawnIn += 1;
      const ids = ringIds(before);
      for (let i = 0; i < ids.length; i += 1) for (let j = i + 1; j < ids.length; j += 1) {
        const sizes = (l: CapabilityCosmosLayout, ring: number) => [...new Map(l.nodes.filter(n => n.ringIndex === ring).map(n => [`${n.width}x${n.height}`, n])).values()];
        for (const a of sizes(before, ids[i])) for (const b of sizes(before, ids[j])) {
          const only = (l: CapabilityCosmosLayout) => ({ ...l, nodes: l.nodes.filter(n => (n.ringIndex === ids[i] && n.width === a.width && n.height === a.height) || (n.ringIndex === ids[j] && n.width === b.width && n.height === b.height)) });
          const p = acrossInMotion(only(before), ids[i], ids[j], RULE_EXCESS), d = acrossInMotion(only(layout), ids[i], ids[j], RULE_EXCESS);
          if (p >= 0) least = Math.min(least, d);
          if (p >= 0 && d < -TOLERANCE) apart.push(`${count} ${kind} ${width}x${height} ${compactness.x}/${compactness.y} rings ${ids[i]}-${ids[j]} ${a.width}x${a.height}/${b.width}x${b.height}: packed ${p.toFixed(2)} drawn ${d.toFixed(2)}`);
          if (p < 0 && d < p - TOLERANCE) closer.push(`${count} ${kind} ${width}x${height} ${compactness.x}/${compactness.y} rings ${ids[i]}-${ids[j]} ${a.width}x${a.height}/${b.width}x${b.height}: packed ${p.toFixed(2)} drawn ${d.toFixed(2)}`);
        }
      }
    }
    console.log(`O-11 moving layouts with more than one orbit: ${multi}; drawn in: ${drawnIn}; least air where the packing held ${least.toFixed(2)} px; air broken where the packing held: ${apart.length}; closer than the packing: ${closer.length}\n${[...apart, ...closer].slice(0, 30).join("\n")}`);
    expect(drawnIn).toBeGreaterThan(0);
    expect({ apart, closer }).toEqual({ apart: [], closer: [] });
  });

  for (const kind of Object.keys(TITLES)) for (const height of HEIGHTS) it(`O-6 smallest orbit, one step (${kind}, height ${height}): a step of 0.01 tighter on either axis breaks an air (rule card height, confirmed every 0.01°) or stands the outer orbit up`, async () => {
    await idle();
    const report: string[] = [];
    let checked = 0, confirmed = 0;
    for (const r of rows) {
      if (r.kind !== kind || !r.box.endsWith(`x${height}`) || r.before.staticPositions) continue;
      const state = packedState(r);
      for (const [kx, ky] of [[r.kx - 0.01, r.ky], [r.kx, r.ky - 0.01]]) {
        if (kx <= 0.005 || ky <= 0.005) continue;
        checked += 1;
        if (!lies(r, kx, ky)) continue;
        if (ruleAir(r.before, kx, ky, state) < HOLDS_MARGIN) continue;
        confirmed += 1;
        const m = ruleAir(r.before, kx, ky, state, CONFIRM, true);
        if (m >= HOLDS_MARGIN) report.push(`${r.count} ${r.kind} ${r.box} (${r.rings} orbits) at ${r.kx}/${r.ky}: ${kx.toFixed(2)}/${ky.toFixed(2)} still holds (${m.toFixed(3)} px at 0.01°)`);
      }
    }
    console.log(`O-6 ${kind} height ${height}: tighter steps ${checked}; read again every 0.01°: ${confirmed}; still holding: ${report.length}${report.length ? `\n${report.join("\n")}` : ""}`);
    expect(report).toEqual([]);
  });

  for (const kind of ["real", "random_4711", "long"]) it(`O-7 smallest orbit, every horizontal factor (subset, ${kind}): no pair of factors that holds and lies gives a smaller kx·ky than the chosen one (confirmed every 0.01°)`, async () => {
    await idle();
    const subset = rows.filter(r => !r.before.staticPositions && r.kind === kind && [1, 2, 3, 5, 8].includes(r.count) && ["1600x740", "1280x560", "1366x620", "960x420", "2560x960"].includes(r.box));
    const report: string[] = [], table: string[] = [];
    for (const r of subset) {
      const state = packedState(r);
      const holds = (sx: number, sy: number, phases = FINE) => ruleAir(r.before, sx / 100, sy / 100, state, phases, phases === CONFIRM) >= HOLDS_MARGIN;
      const chosen = Math.round(r.kx * 100) * Math.round(r.ky * 100);
      let best = { sx: 100, sy: 100 };
      for (let sx = 1; sx <= 100; sx += 1) {
        if (!holds(sx, 100)) continue;
        let low = 1, high = 100;
        while (low < high) { const mid = (low + high) >> 1; if (holds(sx, mid)) high = mid; else low = mid + 1; }
        // a smaller orbit than the chosen one is read again every 0.01° (from its vertical factor up)
        if (sx * low < chosen) while (low <= 100 && sx * low < chosen && !holds(sx, low, CONFIRM)) low += 1;
        if (low > 100 || !lies(r, sx / 100, low / 100)) continue;
        if (sx * low < best.sx * best.sy) best = { sx, sy: low };
      }
      table.push(`${r.count} ${r.kind} ${r.box}: chosen ${r.kx}/${r.ky} (${chosen}) PINK least ${best.sx / 100}/${best.sy / 100} (${best.sx * best.sy})`);
      if (best.sx * best.sy < chosen) report.push(table[table.length - 1]);
    }
    console.log(`O-7 ${kind}: subset ${subset.length} layouts; a smaller holding orbit found: ${report.length}\n${table.join("\n")}`);
    expect(subset.length).toBeGreaterThan(20);
    expect(report).toEqual([]);
  });

  it("O-8 the same cards give the same drawn orbit over every box that can carry it (packed radii at least the least orbit's): outer orbit area within the hundredth grid of the factors", async () => {
    await idle();
    const groups = new Map<string, Row[]>();
    for (const r of drawn) if (r.rings === 1) { const key = `${r.kind} ${r.count} ${r.after.nodes.map(n => `${n.width}x${n.height}@${n.angle.toFixed(4)}`).join(",")}`; groups.set(key, [...(groups.get(key) ?? []), r]); }
    const report: string[] = [], spreads: Array<{ k: string; n: number; spread: number; allowed: number }> = [];
    for (const [k, group] of groups) {
      if (group.length < 2) continue;
      const area = (r: Row) => { const o = outerOf(r.after); return o.radiusX * o.radiusY; };
      const best = group.reduce((m, r) => (area(r) < area(m) ? r : m)), bo = outerOf(best.after);
      const reach = group.filter(r => { const o = outerOf(r.before); return o.radiusX >= bo.radiusX && o.radiusY >= bo.radiusY; });
      let spread = 0, allowedMax = 0;
      for (const r of reach) {
        const o = outerOf(r.before);
        const allowed = (0.01 * o.radiusX) / bo.radiusX + (0.01 * o.radiusY) / bo.radiusY + 0.005;
        const over = area(r) / area(best) - 1;
        spread = Math.max(spread, over); allowedMax = Math.max(allowedMax, allowed);
        if (over > allowed) report.push(`${k.slice(0, 70)} at ${r.box} ${r.kx}/${r.ky}: +${(over * 100).toFixed(2)} % > ${(allowed * 100).toFixed(2)} %`);
      }
      spreads.push({ k, n: reach.length, spread, allowed: allowedMax });
    }
    spreads.sort((a, b) => b.spread - a.spread);
    console.log(`O-8 single-orbit card sets drawn in at more than one box: ${spreads.length}; widest spread over the boxes that can carry the least orbit +${(spreads[0].spread * 100).toFixed(2)} %; above the hundredth grid: ${report.length}\n${spreads.slice(0, 6).map(x => `  +${(x.spread * 100).toFixed(2)} % (grid ${(x.allowed * 100).toFixed(2)} %) over ${x.n} boxes: ${x.k.replace(/@[-0-9.]+/g, "").slice(0, 80)}`).join("\n")}\n${report.slice(0, 12).join("\n")}`);
    expect(spreads.length).toBeGreaterThan(20);
    expect(report).toEqual([]);
  });

  it("O-9 cost and distribution (measured)", async () => {
    await idle();
    const sorted = [...times].sort((a, b) => a - b);
    const dist = new Map<string, number>(); for (const r of drawn) dist.set(`${r.rings} orbit(s)`, (dist.get(`${r.rings} orbit(s)`) ?? 0) + 1);
    const multi = rows.filter(r => r.rings > 1 && !r.before.staticPositions);
    console.log(`O-9 syntxCosmosLayout over ${rows.length} layouts: median ${sorted[Math.floor(sorted.length / 2)].toFixed(2)} ms, p99 ${sorted[Math.floor(sorted.length * 0.99)].toFixed(1)} ms, slowest ${sorted[sorted.length - 1].toFixed(1)} ms`);
    console.log(`O-9 drawn in: ${drawn.length} of ${rows.length} (${[...dist].map(([k, v]) => `${k}: ${v}`).join(", ")}); moving layouts with more than one orbit: ${multi.length}, drawn in ${multi.filter(r => r.kx !== 1 || r.ky !== 1).length}; static ${rows.filter(r => r.before.staticPositions).length}`);
    const kx = drawn.map(r => r.kx), ky = drawn.map(r => r.ky);
    console.log(`O-9 factors drawn in: kx ${Math.min(...kx)}–${Math.max(...kx)}, ky ${Math.min(...ky)}–${Math.max(...ky)}`);
  });

  it("O-10 Capability Field untouched: the shared L1 packing called directly is the layout syntxCosmosPacking returns for fitting items", async () => {
    await idle();
    const items = roles(5, "real");
    expect(buildOrbitalCosmosLayout({ items: [...items], width: 1600, height: 740 })).toEqual(syntxCosmosPacking(items, 1600, 740).layout);
  });
