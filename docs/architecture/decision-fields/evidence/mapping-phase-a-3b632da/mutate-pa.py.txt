for mid in sys.argv[2:]:
    label, f, old, new = M[mid]; W = f"{INT}/tpa-{mid}"
    if os.path.exists(W): shutil.rmtree(W)
    shutil.copytree(SRC, W, symlinks=True, ignore=shutil.ignore_patterns("node_modules", ".next", "test/pink"))
    os.symlink(os.path.realpath(f"{SRC}/node_modules"), f"{W}/node_modules")
    p = f"{W}/{f}"; s = open(p).read()
    if s.count(old) != 1: print(f"{mid} PATTERN {s.count(old)} {label}", flush=True); shutil.rmtree(W); continue
    open(p, "w").write(s.replace(old, new))
    try:
        r = subprocess.run(["npm", "run", "-s", "test:isolated", "--", *T], cwd=W, capture_output=True, text=True, timeout=900)
        out = r.stdout + r.stderr; tests = [l.strip() for l in out.splitlines() if l.strip().startswith("Tests ")]
        fails = sorted({l.split(">")[-1].strip()[:70] for l in out.splitlines() if l.strip().startswith("FAIL ")})
        print(f"{mid} {'RED' if r.returncode != 0 else 'GREEN'} {label} | {tests[-1] if tests else 'no summary'} | {fails[:2]}", flush=True)
    except subprocess.TimeoutExpired: print(f"{mid} TIMEOUT {label}", flush=True)
    shutil.rmtree(W)
