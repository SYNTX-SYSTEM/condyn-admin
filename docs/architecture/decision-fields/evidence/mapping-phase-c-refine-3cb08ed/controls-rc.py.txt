# PINK §3.99 positive controls for the two architecture tests on 3cb08edf, before and after a simulated merge. argv: <pristine msrc2> <work root>
# Each control: hard-linked copy of msrc2 (cp -al), changes written to NEW inodes (os.replace; git writes refs/index by lock+rename),
# the two architecture tests run. Scratch commits use a scratch identity, are unsigned and never leave the copy. Nothing is removed.
import os, subprocess, sys
SRC, ROOT = sys.argv[1], sys.argv[2]
RC = "3cb08edfa3fa6c8e552739380bdc52499076f3f4"
TESTS = ["test/career/mapping/architecture.test.ts", "test/career/mapping-field/architecture.test.ts"]
GIT = ["git", "-c", "user.name=PINK scratch", "-c", "user.email=pink@scratch.invalid", "-c", "commit.gpgsign=false"]
def write_new(path, text):
    tmp = f"{path}.pink-control"
    with open(tmp, "w") as f: f.write(text)
    os.replace(tmp, path)
def append(rel, line): return lambda W: write_new(f"{W}/{rel}", open(f"{W}/{rel}").read() + line)
def create(rel, text): return lambda W: write_new(f"{W}/{rel}", text)
def old_arch(W): write_new(f"{W}/test/career/mapping-field/architecture.test.ts", subprocess.run(["git", "show", "fa895193:test/career/mapping-field/architecture.test.ts"], cwd=W, capture_output=True, text=True, check=True).stdout)
def merge(W): subprocess.run(["git", "update-ref", "refs/remotes/origin/main", RC], cwd=W, check=True)
def commit_on_main(*changes):
    # main moves on after the merge: the changes are committed in the copy and origin/main := that commit (merge base = HEAD)
    def run(W):
        for c in changes: c(W)
        subprocess.run([*GIT, "commit", "-q", "-a", "-m", "pink scratch: main moves on"], cwd=W, check=True)
        subprocess.run(["git", "update-ref", "refs/remotes/origin/main", "HEAD"], cwd=W, check=True)
    return run
def seq(*fs): return lambda W: [f(W) for f in fs]
IMPORTER = create("lib/career/pink-importer.ts", 'import { buildMappingView } from "./mapping/view";\nexport const pinkImporter = buildMappingView;\n')
ENGINE = append("lib/career/mapping/view.ts", "\n// pink control\n")
ORBIT = append("app/components/career/demo/ResonanceOrbits.tsx", "\n// pink control\n")
UNLISTED = append("app/components/career/demo/RoleDiscoveryPanel.tsx", "\n// pink control\n")
SELECTOR = append("app/components/career/demo/SyntxFieldSelector.tsx", "\n// pink control\n")
PANEL = append("app/components/career/demo/AutoJobRankingPanel.tsx", "\n// pink control\n")
UNRELATED = append("README.md", "\n<!-- pink control: unrelated later change on main -->\n")
K = {
 "K0":  ("unchanged, before merge (must be green)", lambda W: None),
 "K1":  ("before merge: new lib/career file imports the engine as ./mapping/view", IMPORTER),
 "K2":  ("before merge: engine file changed", ENGINE),
 "K3":  ("before merge: protected orbit changed", ORBIT),
 "K4":  ("before merge: unlisted existing file changed", UNLISTED),
 "K0m": ("after merge, unchanged (F-PC-1: must be green)", merge),
 "K7m": ("after merge, fa895193's old mapping-field architecture test (expect exactly the 2 F-PC-1 reds)", seq(merge, old_arch)),
 "K1m": ("after merge: ./mapping/view importer", seq(merge, IMPORTER)),
 "K2m": ("after merge: engine file changed", seq(merge, ENGINE)),
 "K3m": ("after merge: protected orbit changed", seq(merge, ORBIT)),
 "K4m": ("after merge: unlisted existing file changed", seq(merge, UNLISTED)),
 "K8m": ("after merge: SyntxFieldSelector + AutoJobRankingPanel changed (GRÜN's probe)", seq(merge, SELECTOR, PANEL)),
 "K9c": ("main moves on: unrelated committed change (README) after the merge (phase boundary)", commit_on_main(UNRELATED)),
 "K10c": ("main moves on: committed engine change after the merge (O-PB-3)", commit_on_main(ENGINE)),
}
for kid, (label, apply) in K.items():
    W = f"{ROOT}/k-{kid}"
    if os.path.exists(W): print(f"{kid} EXISTS {W} (not reused)", flush=True); continue
    subprocess.run(["cp", "-al", SRC, W], check=True)
    apply(W)
    state = subprocess.run(["git", "log", "-1", "--format=%h", "HEAD"], cwd=W, capture_output=True, text=True).stdout.strip() + " main=" + subprocess.run(["git", "rev-parse", "--short", "origin/main"], cwd=W, capture_output=True, text=True).stdout.strip()
    r = subprocess.run(["npm", "run", "-s", "test:isolated", "--", *TESTS], cwd=W, capture_output=True, text=True, timeout=600)
    out = r.stdout + r.stderr
    tests = [l.strip() for l in out.splitlines() if l.strip().startswith("Tests ")]
    fails = sorted({l.strip()[l.strip().find(" ") + 1:][:150] for l in out.splitlines() if l.strip().startswith("× ") or l.strip().startswith("✗ ") or (l.strip().startswith("FAIL ") and ">" in l)})
    print(f"{kid} {'RED' if r.returncode != 0 else 'GREEN'} [{state}] {label} | {tests[-1] if tests else 'no summary'} | {fails}", flush=True)
    with open(f"{ROOT}/k-{kid}.out.txt", "w") as f: f.write(out)
