// ---------- seeded randomness (test side only) ----------
function rngOf(seed: number) { let a = seed >>> 0; return () => { a = (a + 0x6d2b79f5) >>> 0; let t = a; t = Math.imul(t ^ (t >>> 15), t | 1); t ^= t + Math.imul(t ^ (t >>> 7), t | 61); return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }
type R = () => number;
const int = (r: R, lo: number, hi: number) => lo + Math.floor(r() * (hi - lo + 1));
const pick = <T,>(r: R, xs: readonly T[]): T => xs[Math.floor(r() * xs.length)];
const shuffle = <T,>(r: R, xs: readonly T[]): T[] => { const a = [...xs]; for (let i = a.length - 1; i > 0; i -= 1) { const j = Math.floor(r() * (i + 1)); [a[i], a[j]] = [a[j], a[i]]; } return a; };
const clone = <T,>(v: T): T => JSON.parse(JSON.stringify(v));
const text = (a: string, b: string) => (a < b ? -1 : a > b ? 1 : 0);
function freezeDeep<T>(v: T): T { if (v && typeof v === "object") { for (const x of Object.values(v)) freezeDeep(x); Object.freeze(v); } return v; }
function anyFrozen(v: unknown): boolean { return !!v && typeof v === "object" && (Object.isFrozen(v) || Object.values(v).some(anyFrozen)); }
function deepFrozen(v: unknown): boolean { return !v || typeof v !== "object" || (Object.isFrozen(v) && Object.values(v).every(deepFrozen)); }

const ANALYSIS = "CAN_PINK0000000000000000000000000PB";
const W = { REQUIRED: 3, PREFERRED: 2, OPTIONAL: 1, UNKNOWN: 1 } as const;
const SV = { EQUIVALENT: 1, COVERS: 1, PARTIAL: 0.5, DIFFERENT: 0, UNCLEAR: 0 } as const;
const LV = { NAME_EXACT: 0.5, NAME_ALIAS: 0.5, NAME_COMPOSITE_CONSTITUENT: 0, WEAK_TOKEN_CONTAINMENT: 0 } as const;
const NECESSITIES = ["REQUIRED", "PREFERRED", "OPTIONAL", "UNKNOWN"] as const;
const CAP_NAMES = ["TypeScript", "PostgreSQL", "Kubernetes", "Python", "Go", "Rust"];
type Person = Array<{ key: string; name: string }>;
function personOf(r: R): Person {
  const n = int(r, 0, 6); const out: Person = [];
  for (let i = 0; i < n; i += 1) out.push({ key: `PCAP_P${i}_${int(r, 0, 999)}`, name: r() < 0.25 && out.length ? out[0].name : pick(r, CAP_NAMES) });
  return out;
}
type KeyMode = "KNOWN" | "FOREIGN" | "NULL" | "ABSENT";
function requirementOf(r: R, id: string, person: Person, stats: Record<string, number>): RankedRequirement {
  const necessity = pick(r, NECESSITIES), weight = W[necessity];
  const roll = r();
  const source = roll < 0.5 ? "SEMANTIC" : roll < 0.8 ? "LEXICAL" : "NONE";
  const quote = [{ sourceDocumentId: "DOC_PINK", exactQuote: `placeholder ${id}` }];
  if (source === "NONE") {
    const q: Record<string, unknown> = { poolRequirementId: id, requirementText: `Req ${id}`, necessity, weight, source, state: "NONE", value: 0, contribution: 0, capabilityName: null, explanation: null, evidenceLimitation: null, quotes: [] };
    if (r() < 0.5) q.capabilityKey = null;
    return q as unknown as RankedRequirement;
  }
  const mode: KeyMode = pick(r, ["KNOWN", "KNOWN", "FOREIGN", "NULL", "ABSENT", "ABSENT"] as const);
  const cap = person.length ? pick(r, person) : null;
  // The name trap: a key-less coupling often carries exactly a PERSON capability's name.
  const name = mode === "KNOWN" && cap ? (r() < 0.8 ? cap.name : "Label differs from PERSON name") : cap && r() < 0.7 ? cap.name : pick(r, CAP_NAMES);
  const key = mode === "KNOWN" && cap ? cap.key : mode === "FOREIGN" ? `PCAP_FOREIGN_${int(r, 0, 3)}` : mode === "NULL" ? null : mode === "KNOWN" ? null : undefined;
  stats[`key_${mode === "KNOWN" && !cap ? "NULL" : mode}`] = (stats[`key_${mode === "KNOWN" && !cap ? "NULL" : mode}`] ?? 0) + 1;
  let q: Record<string, unknown>;
  if (source === "SEMANTIC") {
    const state = pick(r, Object.keys(SV) as Array<keyof typeof SV>), value = SV[state];
    q = { poolRequirementId: id, requirementText: `Req ${id}`, necessity, weight, source, state, value, contribution: weight * value, capabilityName: name,
      explanation: `Statement ${id}`, evidenceLimitation: `Limit ${id}`, quotes: state === "DIFFERENT" || state === "UNCLEAR" ? (r() < 0.5 ? [] : quote) : quote };
  } else {
    const state = pick(r, Object.keys(LV) as Array<keyof typeof LV>), value = LV[state];
    q = { poolRequirementId: id, requirementText: `Req ${id}`, necessity, weight, source, state, value, contribution: weight * value, capabilityName: name, explanation: null, evidenceLimitation: null, quotes: r() < 0.5 ? quote : [] };
  }
  if (key !== undefined) q.capabilityKey = key;
  return q as unknown as RankedRequirement;
}
function rankingOf(r: R, person: Person, stats: Record<string, number>, maxRoles = 14): RoleRanking {
  const poolCount = int(r, 1, 4);
  const pools = Array.from({ length: poolCount }, (_, i) => ({ jobPoolUploadId: `JPOOL_PINK_${i}_${int(r, 0, 99)}`, poolName: pick(r, ["Alpha", "Beta", "Gamma", "Alpha"]) }));
  const used = new Set<string>(); const built: Array<Omit<RankedRole, "rank">> = [];
  for (let i = 0, n = int(r, 0, maxRoles); i < n; i += 1) {
    const pool = pick(r, pools), poolRoleId = `role_${int(r, 0, 9)}`;
    if (used.has(`${pool.jobPoolUploadId}/${poolRoleId}`)) continue; used.add(`${pool.jobPoolUploadId}/${poolRoleId}`);
    const requirements = Array.from({ length: int(r, 0, 6) }, (_, k) => requirementOf(r, `q_${k}_${int(r, 0, 9)}`, person, stats)).filter((q, k, all) => all.findIndex(x => x.poolRequirementId === q.poolRequirementId) === k);
    const weights = requirements.reduce((s, q) => s + q.weight, 0);
    const evaluated = requirements.filter(q => q.source === "SEMANTIC").length;
    built.push({ tier: evaluated ? "SEMANTIC" : "LEXICAL_ONLY", fit: requirements.length ? requirements.reduce((s, q) => s + q.contribution, 0) / weights : null,
      requiredGaps: requirements.filter(q => q.necessity === "REQUIRED" && q.value === 0).length, semanticCoverage: { evaluated, total: requirements.length },
      role: { jobPoolUploadId: pool.jobPoolUploadId, poolRoleId, title: pick(r, ["Engineer", "Lead", "Analyst"]), organizationName: "Org", poolName: pool.poolName }, requirements });
  }
  built.sort((a, b) => (a.tier === b.tier ? 0 : a.tier === "SEMANTIC" ? -1 : 1) || (a.fit === null ? 1 : 0) - (b.fit === null ? 1 : 0) || (b.fit ?? 0) - (a.fit ?? 0) || a.requiredGaps - b.requiredGaps
    || text(a.role.title, b.role.title) || text(a.role.jobPoolUploadId, b.role.jobPoolUploadId) || text(a.role.poolRoleId, b.role.poolRoleId));
  const ranks = built.map((x, at) => ({ rank: at + 1, ...x }));
  const body = { rankingVersion: "ROLE_RANKING_V1", authority: { ranking: true, decision: false, authorityState: "NONE" }, analysisId: ANALYSIS,
    evidenceClass: ranks.some(x => x.tier === "SEMANTIC") ? "STAND_IN" : null, pools, poolsDropped: [], formula: formula(), counts: { pools: pools.length, roles: ranks.length, ranked: ranks.length }, ranks };
  const decoded = decodeRoleRanking(clone(body), ANALYSIS);
  if (decoded === null) throw new Error("PINK generator produced a body the strict decoder refuses");
  return decoded;
}

function expectedView(person: Person, ranking: RoleRanking) {
  const keys = new Set(person.map(p => p.key));
  const order = (q: RankedRequirement) => NECESSITIES.indexOf(q.necessity as never);
  const roles = [...ranking.ranks].sort((a, b) => a.rank - b.rank);
  const couplings: unknown[] = [];
  const outRoles = roles.map(x => {
    const reqs = [...x.requirements].sort((a, b) => order(a) - order(b) || b.weight - a.weight || text(a.poolRequirementId, b.poolRequirementId));
    for (const q of reqs) if (q.capabilityName !== null) {
      const key = Object.hasOwn(q, "capabilityKey") ? q.capabilityKey! : null;
      couplings.push({ role: { jobPoolUploadId: x.role.jobPoolUploadId, poolRoleId: x.role.poolRoleId }, poolRequirementId: q.poolRequirementId, source: q.source, state: q.state,
        capabilityKey: key, capabilityName: q.capabilityName, keyStatus: key === null ? "ABSENT" : keys.has(key) ? "KNOWN" : "NOT_IN_PERSON", explanation: q.explanation, evidenceLimitation: q.evidenceLimitation, quotes: q.quotes });
    }
    return { id: { jobPoolUploadId: x.role.jobPoolUploadId, poolRoleId: x.role.poolRoleId }, title: x.role.title, organizationName: x.role.organizationName, rank: x.rank, tier: x.tier, fit: x.fit,
      weightSum: x.requirements.reduce((s, q) => s + q.weight, 0),
      requirements: reqs.map(q => ({ poolRequirementId: q.poolRequirementId, requirementText: q.requirementText, necessity: q.necessity, weight: q.weight, source: q.source, state: q.state, value: q.value, capabilityKey: Object.hasOwn(q, "capabilityKey") ? q.capabilityKey : null })) };
  });
  const pools = [...ranking.pools].sort((a, b) => text(a.poolName, b.poolName) || text(a.jobPoolUploadId, b.jobPoolUploadId))
    .map(p => ({ id: p.jobPoolUploadId, name: p.poolName, roles: roles.filter(x => x.role.jobPoolUploadId === p.jobPoolUploadId).length }));
  return { roles: outRoles, couplings, pools, constellation: [...person].sort((a, b) => text(a.name, b.name) || text(a.key, b.key)) };
}

      for (const k of view.couplings) { status[k.keyStatus] = (status[k.keyStatus] ?? 0) + 1; couplings += 1; couplingSign(k.source, k.state); }
      // name trap: a key-less coupling carrying a PERSON name never appears under that PERSON key
      for (const k of view.couplings.filter(x => x.capabilityKey === null && person.some(p => p.name === x.capabilityName))) {
        trap += 1;
        for (const p of person.filter(p => p.name === k.capabilityName)) expect(reverseView(view, p.key).couplings.some(x => x.poolRequirementId === k.poolRequirementId && x.role.poolRoleId === k.role.poolRoleId && x.role.jobPoolUploadId === k.role.jobPoolUploadId && x.capabilityKey === null)).toBe(false);
      }
      for (const key of [...person.map(p => p.key), "PCAP_FOREIGN_0", "PCAP_FOREIGN_1", ...CAP_NAMES]) {
        const rv = reverseView(view, key); reverse += 1;
        expect(clone(rv.couplings)).toEqual(clone(view.couplings.filter(x => x.capabilityKey === key)));
        const ids = view.roles.filter(x => rv.couplings.some(k => k.role.jobPoolUploadId === x.id.jobPoolUploadId && k.role.poolRoleId === x.id.poolRoleId)).map(x => x.id);
        expect(clone(rv.roles)).toEqual(clone(ids));
        expect(rv.known).toBe(person.some(p => p.key === key));
        expect(deepFrozen(rv)).toBe(true);
      }
      // any order of the inputs → the same view
      const shuffled = { ...clone(ranking), pools: shuffle(r, ranking.pools), ranks: shuffle(r, ranking.ranks).map(x => ({ ...x, requirements: shuffle(r, x.requirements) })) };
      expect(JSON.stringify(buildMappingView({ capabilities: shuffle(r, person), ranking: shuffled as RoleRanking, run, evidenceClass: ranking.evidenceClass }))).toBe(JSON.stringify(view)); permuted += 1;
      // deep-frozen input works
      expect(JSON.stringify(buildMappingView({ capabilities: freezeDeep(clone(person)), ranking: freezeDeep(clone(ranking)), run, evidenceClass: ranking.evidenceClass }))).toBe(JSON.stringify(view));
    }
    // empty and refusals
    expect(buildMappingView({ capabilities: [], ranking: null, run: { state: "NOT_STARTED", resumePending: false }, evidenceClass: null }).empty).toBe("NO_RANKING");
    expect(() => buildMappingView({ capabilities: [{ key: "K", name: "a" }, { key: "K", name: "b" }], ranking: null, run: { state: "NOT_STARTED", resumePending: false }, evidenceClass: null })).toThrow(/DUPLICATE_CAPABILITY/);
    expect(capabilitiesOfProjection(null)).toEqual([]);
    expect(capabilitiesOfProjection({ capabilities: [{ id: "PCAP_X", name: "X", extra: 1 }] } as never)).toEqual([{ key: "PCAP_X", name: "X" }]);
    rec.view = { views, couplings, keyStatus: status, generatedKeys: stats, nameTrapCouplings: trap, reverseViews: reverse, permutedRuns: permuted };
    expect(trap).toBeGreaterThan(50); expect(status.NOT_IN_PERSON ?? 0).toBeGreaterThan(50); expect(status.ABSENT ?? 0).toBeGreaterThan(50);
    dump();
  }, 300_000);

type Change = ReturnType<typeof diffRankings>[number];
function pinkDiff(a: RoleRanking | null, b: RoleRanking | null): unknown[] {
  const id = (x: RankedRole) => `${x.role.jobPoolUploadId}\u0000${x.role.poolRoleId}`;
  const A = new Map((a?.ranks ?? []).map(x => [id(x), x])), B = new Map((b?.ranks ?? []).map(x => [id(x), x]));
  const k = (q: RankedRequirement) => (Object.hasOwn(q, "capabilityKey") && q.capabilityKey !== undefined ? q.capabilityKey : null);
  const out: Array<{ s: string; c: unknown }> = [];
  const order = ["ROLE_ADDED", "ROLE_REMOVED", "ROLE_RANK_CHANGED", "ROLE_TIER_CHANGED", "ROLE_FIT_CHANGED", "REQUIREMENT_ADDED", "REQUIREMENT_REMOVED", "REQUIREMENT_STATE_CHANGED", "REQUIREMENT_KEY_CHANGED"];
  const push = (role: RankedRole, q: string | null, kind: string, from: unknown, to: unknown) => out.push({ s: `${role.role.jobPoolUploadId}\u0001${role.role.poolRoleId}\u0001${q ?? ""}\u0001${String(Math.min(order.indexOf(kind), kind.endsWith("REMOVED") ? order.indexOf(kind) - 1 : 99)).padStart(2, "0")}`,
    c: { kind, role: { jobPoolUploadId: role.role.jobPoolUploadId, poolRoleId: role.role.poolRoleId }, poolRequirementId: q, from, to } });
  for (const key of [...new Set([...A.keys(), ...B.keys()])].sort()) {
    const x = A.get(key), y = B.get(key);
    if (!x) { push(y!, null, "ROLE_ADDED", null, { rank: y!.rank, tier: y!.tier, fit: y!.fit }); continue; }
    if (!y) { push(x, null, "ROLE_REMOVED", { rank: x.rank, tier: x.tier, fit: x.fit }, null); continue; }
    if (x.rank !== y.rank) push(x, null, "ROLE_RANK_CHANGED", x.rank, y.rank);
    if (x.tier !== y.tier) push(x, null, "ROLE_TIER_CHANGED", x.tier, y.tier);
    if (x.fit !== y.fit) push(x, null, "ROLE_FIT_CHANGED", x.fit, y.fit);
    const QA = new Map(x.requirements.map(q => [q.poolRequirementId, q])), QB = new Map(y.requirements.map(q => [q.poolRequirementId, q]));
    for (const qid of new Set([...QA.keys(), ...QB.keys()])) {
      const p = QA.get(qid), q = QB.get(qid);
      if (!p) { push(x, qid, "REQUIREMENT_ADDED", null, { source: q!.source, state: q!.state, capabilityKey: k(q!) }); continue; }
      if (!q) { push(x, qid, "REQUIREMENT_REMOVED", { source: p.source, state: p.state, capabilityKey: k(p) }, null); continue; }
      if (p.source !== q.source || p.state !== q.state) push(x, qid, "REQUIREMENT_STATE_CHANGED", { source: p.source, state: p.state }, { source: q.source, state: q.state });
      if (k(p) !== k(q)) push(x, qid, "REQUIREMENT_KEY_CHANGED", k(p), k(q));
    }
  }
  return out.sort((m, n) => text(m.s, n.s)).map(x => x.c);
}
const mirror = (c: Change): unknown => ({ ...c, kind: c.kind === "ROLE_ADDED" ? "ROLE_REMOVED" : c.kind === "ROLE_REMOVED" ? "ROLE_ADDED" : c.kind === "REQUIREMENT_ADDED" ? "REQUIREMENT_REMOVED" : c.kind === "REQUIREMENT_REMOVED" ? "REQUIREMENT_ADDED" : c.kind, from: c.to, to: c.from });
function mutateRanking(r: R, a: RoleRanking, person: Person, stats: Record<string, number>): RoleRanking {
  const b = clone(a) as RoleRanking; const ops: string[] = [];
  for (let n = int(r, 0, 4); n > 0; n -= 1) {
    const op = pick(r, ["state", "source", "key", "keyAbsentNull", "reqAdd", "reqDel", "roleAdd", "roleDel", "rankSwap", "fit", "tier", "text"]);
    const role = b.ranks.length ? pick(r, b.ranks) : null; const q = role?.requirements.length ? pick(r, role.requirements) : null;
    ops.push(op);
    if (op === "state" && q) q.state = (q.source === "SEMANTIC" ? pick(r, Object.keys(SV)) : q.source === "LEXICAL" ? pick(r, Object.keys(LV)) : "NONE") as never;
    if (op === "source" && q) { q.source = pick(r, ["SEMANTIC", "LEXICAL", "NONE"]); q.state = q.source === "NONE" ? "NONE" : q.source === "SEMANTIC" ? "UNCLEAR" : "NAME_EXACT"; }
    if (op === "key" && q) { const m = pick(r, [0, 1, 2]); if (m === 0) q.capabilityKey = `PCAP_NEW_${int(r, 0, 2)}`; else if (m === 1) q.capabilityKey = null; else delete q.capabilityKey; }
    if (op === "keyAbsentNull" && q) { if (q.capabilityKey === null) delete q.capabilityKey; else if (!Object.hasOwn(q, "capabilityKey")) q.capabilityKey = null; }
    if (op === "reqAdd" && role) role.requirements.push(requirementOf(r, `q_new_${int(r, 0, 99)}`, person, stats));
    if (op === "reqDel" && role && role.requirements.length) role.requirements.splice(int(r, 0, role.requirements.length - 1), 1);
    if (op === "roleAdd") { const extra = rankingOf(r, person, stats, 3).ranks[0]; if (extra && !b.ranks.some(x => x.role.jobPoolUploadId === extra.role.jobPoolUploadId && x.role.poolRoleId === extra.role.poolRoleId)) b.ranks.push({ ...extra, rank: b.ranks.length + 1 }); }
    if (op === "roleDel" && b.ranks.length) b.ranks.splice(int(r, 0, b.ranks.length - 1), 1);
    if (op === "rankSwap" && b.ranks.length > 1) { const i = int(r, 0, b.ranks.length - 1), j = int(r, 0, b.ranks.length - 1); [b.ranks[i].rank, b.ranks[j].rank] = [b.ranks[j].rank, b.ranks[i].rank]; }
    if (op === "fit" && role) role.fit = pick(r, [null, 0, 1, Math.round(r() * 1000) / 1000]);
    if (op === "tier" && role) role.tier = role.tier === "SEMANTIC" ? "LEXICAL_ONLY" : "SEMANTIC";
    if (op === "text" && role) { role.role.title = "Renamed"; if (q) { q.requirementText = "changed text"; q.quotes = []; } }
  }
  for (const o of ops) stats[`op_${o}`] = (stats[`op_${o}`] ?? 0) + 1;
  return b;
}

describe("PINK B · delta", () => {
  it("random pairs: product = PINK's own diff; empty on equality (also shuffled, also key absent vs null); mirror; order-free; inputs untouched", () => {
    const r = rngOf(0xde17); const stats: Record<string, number> = {}; const kinds: Record<string, number> = {};
    let pairs = 0, emptyChecks = 0, smallFit = 0;
    for (let c = 0; c < 500; c += 1) {
      const person = personOf(r); const a = rankingOf(r, person, stats); const b = mutateRanking(r, a, person, stats);
      const sa = JSON.stringify(a), sb = JSON.stringify(b);
      const d = diffRankings(a, b); pairs += 1;
      expect(JSON.parse(JSON.stringify(d))).toEqual(pinkDiff(a, b));
      expect(JSON.stringify(a)).toBe(sa); expect(JSON.stringify(b)).toBe(sb);
      expect(JSON.parse(JSON.stringify(diffRankings(b, a)))).toEqual(JSON.parse(JSON.stringify(d.map(mirror))));
      const sh = (x: RoleRanking) => ({ ...clone(x), ranks: shuffle(r, x.ranks).map(y => ({ ...y, requirements: shuffle(r, y.requirements) })) }) as RoleRanking;
      expect(JSON.stringify(diffRankings(sh(a), sh(b)))).toBe(JSON.stringify(d));
      expect(diffRankings(a, sh(a))).toEqual([]); expect(diffRankings(a, clone(a))).toEqual([]);
      const flipped = clone(a); for (const x of flipped.ranks) for (const q of x.requirements) { if (q.capabilityKey === null) delete q.capabilityKey; else if (!Object.hasOwn(q, "capabilityKey")) q.capabilityKey = null; }
      expect(diffRankings(a, flipped)).toEqual([]); emptyChecks += 3;
      if (a.ranks.length) { const f = clone(a); f.ranks[0].fit = (f.ranks[0].fit ?? 0) + 0.001; expect(diffRankings(a, f).map(x => x.kind)).toEqual(["ROLE_FIT_CHANGED"]); smallFit += 1; }
      for (const x of d) kinds[x.kind] = (kinds[x.kind] ?? 0) + 1;
      expect(diffRankings(null, a).every(x => x.kind === "ROLE_ADDED")).toBe(true);
      expect(diffRankings(null, a)).toHaveLength(a.ranks.length);
    }
    expect(diffRankings(null, null)).toEqual([]);
    rec.delta = { pairs, emptyChecks, smallFitChanges: smallFit, kinds, ops: Object.fromEntries(Object.entries(stats).filter(([k]) => k.startsWith("op_"))) };
    expect(Object.keys(kinds).sort()).toEqual(["REQUIREMENT_ADDED", "REQUIREMENT_KEY_CHANGED", "REQUIREMENT_REMOVED", "REQUIREMENT_STATE_CHANGED", "ROLE_ADDED", "ROLE_FIT_CHANGED", "ROLE_RANK_CHANGED", "ROLE_REMOVED", "ROLE_TIER_CHANGED"]);
    dump();
  }, 300_000);

    const seen = new Map<string, FocusState>([[JSON.stringify(FOCUS_INITIAL), FOCUS_INITIAL]]); const queue: FocusState[] = [FOCUS_INITIAL];
    let transitions = 0, sameObject = 0, invalidSame = 0;

      expect(Object.isFrozen(s)).toBe(true); if (s.role) expect(Object.isFrozen(s.role)).toBe(true); if (s.selection) expect(deepFrozen(s.selection)).toBe(true);
