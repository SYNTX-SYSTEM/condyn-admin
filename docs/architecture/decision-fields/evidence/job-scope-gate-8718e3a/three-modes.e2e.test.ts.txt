async function availablePort(): Promise<number> {
  const listener = createServer();
  await new Promise<void>((ok) => listener.listen(0, "127.0.0.1", () => ok()));
  const address = listener.address() as { port: number };
  await new Promise<void>((ok) => listener.close(() => ok()));
  return address.port;
}

    }
    const urlCases: Array<{ label: string; analysisId: string; pool: string; mode?: string; expect: string; compares: boolean }> = [
      { label: "own PERSON_X_JOB analysis, own pool", analysisId: ANALYSIS_ID, pool: uploadId, expect: "PERSON_X_JOB", compares: true },
      { label: "jobScope-only declaration (earlier contract), own pool", analysisId: URL_LEGACY_DECLARED, pool: uploadId, expect: "PERSON_X_JOB", compares: true },
      { label: "PERSON analysis from the URL", analysisId: URL_PERSON, pool: uploadId, expect: "PERSON", compares: false },
      { label: "analysis row without a job (NOT_RECORDED)", analysisId: URL_NO_JOB, pool: uploadId, expect: "UNRESOLVED", compares: false },
      { label: "unknown analysis (404)", analysisId: URL_UNKNOWN, pool: uploadId, expect: "UNRESOLVED", compares: false },
      { label: "own analysis viewed in another pool", analysisId: ANALYSIS_ID, pool: secondUploadId, expect: "OTHER_POOL", compares: false },
      { label: "own analysis in PERSON mode", analysisId: ANALYSIS_ID, pool: uploadId, mode: "PERSON", expect: "", compares: false },
      { label: "own analysis in JOB mode", analysisId: ANALYSIS_ID, pool: uploadId, mode: "JOB", expect: "", compares: false }
    ];
    const urlEvidence: Record<string, unknown> = {};
    for (const entry of urlCases) {
      const read = await boundaryPage({ focus: "FIELD", analysisId: entry.analysisId, jobPoolUploadId: entry.pool, ...(entry.mode ? { investigationMode: entry.mode } : {}) });
      const matches = matchesOf(read.from); const sources = sourcesOf(read.from);
      urlEvidence[entry.label] = { boundary: read.boundary, fieldState: read.fieldState, hint: read.hintState, hintPool: read.hintPool, matches, sources };
      const comparedOk = entry.compares ? matches.length >= 1 && matches.every((item) => item.includes(`/job-pools/${entry.pool}/`) && item.includes(entry.analysisId)) && read.fieldState === "AVAILABLE" : matches.length === 0 && sources === 0 && read.fieldState !== "AVAILABLE";
      check("T-10", `${entry.label}: boundary ${entry.expect || "(not PERSON × JOB)"}, ${entry.compares ? "compared with exactly its pool" : "no /matches, no /sources, no comparison"}`, (entry.mode ? true : read.boundary === entry.expect) && comparedOk && (entry.compares || entry.mode ? read.hintState === null : read.hintState === entry.expect), urlEvidence[entry.label]);
      if (entry.expect === "OTHER_POOL") {
        check("T-10", "OTHER_POOL names the analysis's own pool and does not switch to it by itself", read.hintPool === uploadId && new URL(read.page.url()).searchParams.get("jobPoolUploadId") === secondUploadId, { hintPool: read.hintPool, url: read.page.url() });
        const fromClick = requests.length;
        await read.page.getByTestId("syntx-analysis-pool-open").click();
        let after: string | null = null;
        for (let attempt = 0; attempt < 120; attempt += 1) { after = await read.page.getByTestId("syntx-field").getAttribute("data-field-state").catch(() => null); if (after === "AVAILABLE") break; await delay(250); }
        const clickMatches = matchesOf(fromClick);
        urlEvidence["OTHER_POOL click"] = { url: read.page.url(), fieldState: after, matches: clickMatches, boundary: await read.page.getByTestId("syntx-field").getAttribute("data-analysis-boundary") };
        check("T-10", "the explicit click opens the analysis's own pool and only then compares it", new URL(read.page.url()).searchParams.get("jobPoolUploadId") === uploadId && after === "AVAILABLE" && clickMatches.length >= 1 && clickMatches.every((item) => item.includes(`/job-pools/${uploadId}/`)), urlEvidence["OTHER_POOL click"]);
        await read.page.screenshot({ path: resolve(EVIDENCE_DIR, `T-10-other-pool-opened-${RUN}.png`) });
      }
      if (entry.expect === "PERSON") await read.page.screenshot({ path: resolve(EVIDENCE_DIR, `T-10-url-person-${RUN}.png`) });
      await read.page.close();
    }
    // A pending boundary read compares nothing; a failed or forged read compares nothing.
    const held: Array<() => void> = [];
    const pending = await (async () => {
      const from = requests.length;
      const page = await browser.newPage({ viewport: { width: 1600, height: 1000 } });
      page.on("request", (request: any) => requests.push({ method: request.method(), url: request.url(), headers: request.headers(), body: request.postData() ?? null }));
      await page.route("**/investigation", async (route: any) => { await new Promise<void>((release) => held.push(release)); await route.continue(); });
      await page.goto(`${base()}${path({ focus: "FIELD", analysisId: ANALYSIS_ID, jobPoolUploadId: uploadId })}`, { waitUntil: "domcontentloaded", timeout: 120_000 });
      await page.getByTestId("field-focus-strip").waitFor({ timeout: 120_000 });
      for (let attempt = 0; attempt < 120 && held.length === 0; attempt += 1) await delay(250);
      await delay(3_000);
      const during = { boundary: await page.getByTestId("syntx-field").getAttribute("data-analysis-boundary").catch(() => null), matches: matchesOf(from).length, sources: sourcesOf(from) };
      held.forEach((release) => release());
      let boundary: string | null = null;
      for (let attempt = 0; attempt < 120; attempt += 1) { boundary = await page.getByTestId("syntx-field").getAttribute("data-analysis-boundary").catch(() => null); if (boundary === "PERSON_X_JOB") break; await delay(250); }
      await delay(1_500);
      const afterRelease = { boundary, matches: matchesOf(from).length };
      await page.close();
      return { during, afterRelease };
    })();
    urlEvidence.pending = pending;
    check("T-10", "while the boundary read is pending: RESOLVING, no /matches, no /sources; after it resolves: PERSON_X_JOB and compared", pending.during.boundary === "RESOLVING" && pending.during.matches === 0 && pending.during.sources === 0 && pending.afterRelease.boundary === "PERSON_X_JOB" && pending.afterRelease.matches >= 1, pending);
    const forged: Array<{ label: string; status: number; body: unknown }> = [
      { label: "server error 500", status: 500, body: { success: false, issues: [{ code: "ERR_ANALYSIS_INVESTIGATION_FAILED" }] } },
      { label: "200 about another analysis", status: 200, body: { success: true, investigation: { investigationVersion: "ANALYSIS_INVESTIGATION_V1", analysisId: "ANL_PINK_OTHER", jobId: "JOB_PINK_OTHER", mode: "PERSON_X_JOB", jobScopeSelection: { jobPoolUploadId: uploadId, poolRoleId: ROLE_ID }, basis: "JOB_DECLARATION", discoveryRunId: null, investigationScopeHash: null } } },
      { label: "200 with an extra key", status: 200, body: { success: true, investigation: { investigationVersion: "ANALYSIS_INVESTIGATION_V1", analysisId: ANALYSIS_ID, jobId: "JOB_PINK_JOB_SCOPE_GATE", mode: "PERSON_X_JOB", jobScopeSelection: { jobPoolUploadId: uploadId, poolRoleId: ROLE_ID }, basis: "JOB_DECLARATION", discoveryRunId: null, investigationScopeHash: null, extra: 1 } } },
      { label: "409 with an unknown reason", status: 409, body: { success: false, issues: [{ code: "ERR_ANALYSIS_INVESTIGATION_UNRESOLVED", reason: "PINK_UNKNOWN" }] } }
    ];
    for (const entry of forged) {
      const read = await boundaryPage({ focus: "FIELD", analysisId: ANALYSIS_ID, jobPoolUploadId: uploadId }, async (page) => { await page.route("**/investigation", (route: any) => route.fulfill({ status: entry.status, contentType: "application/json", body: JSON.stringify(entry.body) })); });
      const matches = matchesOf(read.from);
      urlEvidence[`forged: ${entry.label}`] = { boundary: read.boundary, matches, sources: sourcesOf(read.from) };
      check("T-10", `boundary read ${entry.label}: UNRESOLVED, no /matches, no /sources`, read.boundary === "UNRESOLVED" && matches.length === 0 && sourcesOf(read.from) === 0, urlEvidence[`forged: ${entry.label}`]);
      await read.page.close();
    }
    writeFileSync(resolve(EVIDENCE_DIR, `T-10-url-boundary-${RUN}.json`), `${JSON.stringify(urlEvidence, null, 2)}\n`);

    // ---- T-09 network --------------------------------------------------------------------------------------------------------------
    const foreign = requests.filter((entry) => !entry.url.startsWith(base()) && !entry.url.startsWith("data:") && !entry.url.startsWith("blob:"));
    check("T-09", "the browser sent no request outside the server", foreign.length === 0, foreign.slice(0, 5).map((entry) => entry.url));
    const net = NET_LOG && existsSync(NET_LOG) ? readFileSync(NET_LOG, "utf8").trim().split("\n").filter(Boolean) : [];
    const gemini = net.filter((line) => /generativelanguage|aiplatform|googleapis|gemini/i.test(line));
    check("T-09", `server: 0 Gemini attempts (${SENTINEL && NET_LOG ? "PINK sentinel in the server process" : "no sentinel"}); other refused attempts listed`, gemini.length === 0 && Boolean(SENTINEL && NET_LOG), { total: net.length, gemini: gemini.length, entries: net.slice(0, 5).map((line) => line.split(" ").slice(1).join(" ")) });
    check("T-09", "no page error in the walk", pageErrors.length === 0, pageErrors.slice(0, 5));

    const failed = checks.filter((entry) => !entry.ok);
    console.log(`PINK_THREE_MODES run=${RUN} ${checks.length - failed.length}/${checks.length}`);
    expect(failed.map((entry) => `${entry.id} ${entry.name} · ${entry.detail}`.slice(0, 600))).toEqual([]);
  }, 1_500_000);
});
