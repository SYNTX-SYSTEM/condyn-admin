// PINK §3.82 hit test: does the entry line take pointer hits from the Job Field cosmos nodes? GET-only. argv: <base> <out json> <label>
import { writeFileSync } from "node:fs";
import { pathToFileURL } from "node:url";
const [BASE, OUT, LABEL] = process.argv.slice(2);
const pw = await import(pathToFileURL(process.env.CONDYN_PLAYWRIGHT_MODULE).href); const { chromium } = pw.chromium ? pw : pw.default;
const sleep = ms => new Promise(r => setTimeout(r, ms));
const DLL = "JPOOL_B94850C439974E27496FE3B1A9A076B5", ORG = "org_kestrel", ROLE = "role_kestrel_fullstack";
const Q = "/career/demo?investigationMode=";
const STATES = [["person+poolOpen", `${Q}PERSON&jobPoolUploadId=${DLL}`], ["pxj+poolOpen", `${Q}PERSON_X_JOB&jobPoolUploadId=${DLL}`], ["pxj+roleOpen", `${Q}PERSON_X_JOB&jobPoolUploadId=${DLL}&jobOrganizationId=${ORG}&jobRoleId=${ROLE}`], ["person+roleOpen", `${Q}PERSON&jobPoolUploadId=${DLL}&jobOrganizationId=${ORG}&jobRoleId=${ROLE}`], ["job+poolOpen", `${Q}JOB&jobPoolUploadId=${DLL}`]];
const VPS = [[1920, 1080], [1600, 900], [1440, 900], [1280, 720]];
const b = await chromium.launch(); const out = { label: LABEL, rows: [] };
for (const [W, H] of VPS) for (const [name, path] of STATES) {
  const ctx = await b.newContext({ viewport: { width: W, height: H } }); await ctx.addInitScript(() => { try { localStorage.setItem("condyn_onboarding_seen", "true"); } catch {} });
  let blocked = 0; await ctx.route("**/*", r => { if (!["GET", "HEAD"].includes(r.request().method())) { blocked += 1; return r.abort(); } return r.continue(); });
  const p = await ctx.newPage(); const errors = []; p.on("pageerror", e => errors.push(String(e).slice(0, 120)));
  await p.goto(`${BASE}${path}`, { waitUntil: "domcontentloaded" }); await sleep(7000);
  const geo = await p.evaluate(() => {
    const line = document.querySelector('[data-sx-shell="gate-reason"]'); const L = line?.getBoundingClientRect();
    const nodes = [...document.querySelectorAll('[data-testid^="syntx-organization-"], [data-testid^="syntx-role-"], [data-testid^="syntx-pool"], [data-testid^="syntx-requirement-"]')].filter(e => { const r = e.getBoundingClientRect(); return r.width > 0 && L && r.top < L.bottom && r.bottom > L.top && r.left < L.right && r.right > L.left; });
    return { line: L ? [L.left, L.top, L.width, L.height].map(Math.round) : null, entry: line?.dataset.entryState ?? null, nodes: nodes.map(n => { const r = n.getBoundingClientRect(); let total = 0, toLine = 0, toAction = 0, other = 0;
      for (let i = 1; i < 12; i += 1) for (let j = 1; j < 8; j += 1) { const x = r.left + r.width * i / 12, y = r.top + r.height * j / 8; if (x < 0 || y < 0 || x > innerWidth || y > innerHeight) continue; total += 1; const h = document.elementFromPoint(x, y); if (!h) continue; if (n.contains(h)) continue; if (h.closest('[data-testid="field-entry-action"]')) toAction += 1; else if (h.closest('[data-sx-shell="gate-reason"]')) toLine += 1; else other += 1; }
      const visArea = Math.max(0, Math.min(r.right, L.right) - Math.max(r.left, L.left)) * Math.max(0, Math.min(r.bottom, L.bottom) - Math.max(r.top, L.top));
      return { id: n.dataset.testid, rect: [r.left, r.top, r.width, r.height].map(Math.round), coveredShare: +(visArea / (r.width * r.height)).toFixed(3), samples: total, hitsTakenByAction: toAction, hitsTakenByLine: toLine, hitsTakenByOther: other }; }) };
  });
  const clicks = [];
  for (const n of geo.nodes) { const loc = p.locator(`[data-testid="${n.id}"]`).first(); let trial = "ok"; await loc.click({ trial: true, timeout: 2500 }).catch(e => { trial = /intercepts pointer events/.test(String(e)) ? "INTERCEPTED" : String(e).slice(0, 60); }); clicks.push({ id: n.id, trial }); }
  out.rows.push({ vp: `${W}x${H}`, state: name, ...geo, clicks, blocked, errors });
  console.log(W, name, geo.entry, JSON.stringify(geo.nodes.map(n => [n.id, n.coveredShare, n.hitsTakenByAction, n.hitsTakenByLine])), JSON.stringify(clicks.map(c => c.trial)));
  await ctx.close();
}
await b.close(); writeFileSync(OUT, JSON.stringify(out, null, 1)); console.log("HIT-END", LABEL);
