/**
 * PINK proof of GRÜN df2e84e (three investigation modes, frontend) against GELB's server rule (ad15d7b): the frontend mirror
 * `checkAnalysisPayload` and the request body the real CareerJobController sends, compared with `parseDeclaredInvestigation`
 * over a deterministic, seeded payload space (not a hand-picked list):
 *
 *   M-1  for every payload: frontend accepts ⇔ server accepts; same mode; same refusal code; for PERSON_X_JOB the same selection
 *   M-2  for every accepted payload the controller sends exactly one POST whose body the server reads with the same mode and
 *        selection; a PERSON body carries no `jobScope` key, a PERSON_X_JOB body exactly { jobPoolUploadId, poolRoleId };
 *        every other key of the payload passes unchanged
 *   M-3  for every refused payload the controller sends nothing and keeps its state
 *   M-4  declareAnalysis(mode, validation) for every mode × validation state: PERSON never carries a scope, JOB never analyzes,
 *        PERSON_X_JOB only with VALID — and the declaration is accepted by the server rule
 *   M-5  the URL mode parameter: only the three exact values are a choice
 */
import { describe, expect, it } from "vitest";
import { parseDeclaredInvestigation } from "../../lib/career/investigation/boundary";
import { DEFAULT_INVESTIGATION_MODE, INVESTIGATION_MODES, checkAnalysisPayload, declareAnalysis, investigationModeFromParam } from "../../lib/career/job-pool/investigation-mode";
import type { JobScopeValidation } from "../../lib/career/job-pool/job-scope";
import { CareerJobController } from "../../lib/career/ui/career-job-controller";

const UPLOAD = "JPOOL_6CA8892F9AD0F6AC316F8FC3F6294BB7";
const ROLE = "role_pink_alpha_architect";
const ABSENT = Symbol("absent");

// Deterministic generator (mulberry32): the same 6000 payloads in every run.
function prng(seed: number) { return () => { seed = (seed + 0x6d2b79f5) | 0; let t = Math.imul(seed ^ (seed >>> 15), 1 | seed); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }
const MODE_VALUES: unknown[] = ["PERSON", "PERSON_X_JOB", "JOB", "person", " PERSON", "PERSON ", "PERSON×JOB", "PERSON_X_JOB\u0000", "TEAM", "", null, 0, true, ["PERSON"], { mode: "PERSON" }];
const investigationValues = (pick: () => number): unknown[] => {
  const mode = MODE_VALUES[Math.floor(pick() * MODE_VALUES.length)];
  const proto = Object.create({ mode });
  const nullProto = Object.assign(Object.create(null), { mode });
  return [ABSENT, undefined, null, "PERSON", 1, [], ["PERSON"], {}, { mode }, { mode }, { mode }, { mode, extra: 1 }, { mode, extra: undefined }, { Mode: mode }, proto, nullProto, JSON.parse(`{"__proto__":{"mode":"PERSON"}}`)];
};
const jobScopeValues: unknown[] = [
  ABSENT, undefined, null, "", "JPOOL", 0, [], [UPLOAD, ROLE],
  { jobPoolUploadId: UPLOAD, poolRoleId: ROLE },
  { poolRoleId: ROLE, jobPoolUploadId: UPLOAD },
  { jobPoolUploadId: UPLOAD, poolRoleId: "role pink beta platform" },
  { jobPoolUploadId: UPLOAD, poolRoleId: "  padded  " },
  { jobPoolUploadId: UPLOAD, poolRoleId: "" },
  { jobPoolUploadId: UPLOAD, poolRoleId: 7 },
  { jobPoolUploadId: UPLOAD.toLowerCase(), poolRoleId: ROLE },
  { jobPoolUploadId: UPLOAD.slice(0, -1), poolRoleId: ROLE },
  { jobPoolUploadId: `${UPLOAD}0`, poolRoleId: ROLE },
  { jobPoolUploadId: ` ${UPLOAD}`, poolRoleId: ROLE },
  { jobPoolUploadId: UPLOAD, poolRoleId: ROLE, extra: 1 },
  { jobPoolUploadId: UPLOAD, poolRoleId: ROLE, extra: undefined },
  { jobPoolUploadId: UPLOAD },
  { poolRoleId: ROLE },
  Object.assign(Object.create(null), { jobPoolUploadId: UPLOAD, poolRoleId: ROLE }),
  Object.assign(Object.create({ jobPoolUploadId: UPLOAD }), { poolRoleId: ROLE })
];
function payloads(count: number): unknown[] {
  const pick = prng(20261015);
  const out: unknown[] = [null, undefined, "PERSON", 0, [], [{ investigation: { mode: "PERSON" } }]];
  while (out.length < count) {
    // Half of the space is weighted towards well-formed declarations, so every outcome class is reached many times.
    const investigations = investigationValues(pick);
    const roll = pick();
    const investigation = roll < 0.2 ? ABSENT : roll < 0.5 ? { mode: INVESTIGATION_MODES[Math.floor(pick() * 3)] } : investigations[Math.floor(pick() * investigations.length)];
    const jobScope = pick() < 0.35 ? (pick() < 0.5 ? ABSENT : { jobPoolUploadId: UPLOAD, poolRoleId: ROLE }) : jobScopeValues[Math.floor(pick() * jobScopeValues.length)];
    const payload: Record<string, unknown> = { documents: [{ type: "text", content: `PINK ${out.length}`, title: "t", docId: `DOC_${out.length}` }] };
    if (pick() < 0.15) payload.extraKey = { nested: out.length };
    if (investigation !== ABSENT) payload.investigation = investigation;
    if (jobScope !== ABSENT) payload.jobScope = jobScope;
    out.push(payload);
  }
  return out;
}

type Outcome = { ok: true; mode: string; selection: unknown } | { ok: false; code: string };
function server(payload: unknown): Outcome {
  try {
    const declared = parseDeclaredInvestigation(payload);
    return { ok: true, mode: declared.mode, selection: declared.mode === "PERSON_X_JOB" ? declared.selection : null };
  } catch (error) {
    return { ok: false, code: String((error as { code?: unknown }).code) };
  }
}
function client(payload: unknown): Outcome {
  const checked = checkAnalysisPayload(payload);
  return checked.ok ? { ok: true, mode: checked.mode, selection: "jobScope" in checked.declaration ? checked.declaration.jobScope : null } : { ok: false, code: checked.code };
}

async function send(payload: unknown): Promise<{ result: unknown; bodies: string[]; state: string }> {
  const bodies: string[] = [];
  const fetchFn = (async (_url: string, init: { body: string }) => { bodies.push(init.body); return { status: 400, json: async () => ({ issues: [{ code: "PINK_STOP", message: "stop" }] }) }; }) as unknown as typeof fetch;
  const controller = new CareerJobController(fetchFn);
  const result = await controller.submitAnalysis(payload);
  return { result, bodies, state: controller.getState().state };
}

describe("PINK: three investigation modes — frontend mirror and sent body vs GELB's server rule (df2e84e)", () => {
  const space = payloads(6000);

  it("M-1 frontend and server agree on every payload (acceptance, mode, selection, refusal code)", () => {
    const disagreements: Array<{ index: number; client: Outcome; server: Outcome }> = [];
    const tally: Record<string, number> = {};
    space.forEach((payload, index) => {
      const local = client(payload); const remote = server(payload);
      const key = local.ok ? `ok:${local.mode}` : `refused:${local.code}`;
      tally[key] = (tally[key] ?? 0) + 1;
      if (JSON.stringify(local) !== JSON.stringify(remote)) disagreements.push({ index, client: local, server: remote });
    });
    console.log(`PINK_MIRROR payloads=${space.length} ${Object.entries(tally).sort().map(([k, v]) => `${k}=${v}`).join(" ")} disagreements=${disagreements.length}`);
    // Every outcome class is reached, so agreement is not agreement on one class only.
    for (const key of ["ok:PERSON", "ok:PERSON_X_JOB", "refused:ERR_JOB_SCOPE_REQUIRED", "refused:ERR_INVESTIGATION_INVALID", "refused:ERR_INVESTIGATION_MODE_NOT_ANALYZABLE"]) expect(tally[key] ?? 0).toBeGreaterThan(50);
    expect(disagreements.slice(0, 5)).toEqual([]);
  });

  it("M-2/M-3 the controller sends exactly the checked declaration, read identically by the server; refusals send nothing", async () => {
    let sent = 0, refused = 0;
    const problems: unknown[] = [];
    for (const [index, payload] of space.entries()) {
      const expected = client(payload);
      const { result, bodies, state } = await send(payload);
      if (!expected.ok) {
        refused += 1;
        if (bodies.length !== 0 || result !== expected.code || state !== "IDLE") problems.push({ index, kind: "refusal", bodies: bodies.length, result, state });
        continue;
      }
      sent += 1;
      if (bodies.length !== 1) { problems.push({ index, kind: "count", bodies: bodies.length }); continue; }
      const body = JSON.parse(bodies[0]) as Record<string, unknown>;
      const read = server(body);
      if (JSON.stringify(read) !== JSON.stringify(expected)) problems.push({ index, kind: "server-read", read, expected });
      if (expected.mode === "PERSON" && Object.prototype.hasOwnProperty.call(body, "jobScope")) problems.push({ index, kind: "person-with-scope", body });
      if (expected.mode === "PERSON_X_JOB" && Object.keys(body.jobScope as object).sort().join() !== "jobPoolUploadId,poolRoleId") problems.push({ index, kind: "scope-keys", body });
      if (JSON.stringify(body.investigation) !== JSON.stringify({ mode: expected.mode })) problems.push({ index, kind: "investigation", body });
      // Every other key passes unchanged (documents, extraKey).
      const rest = Object.fromEntries(Object.entries(payload as Record<string, unknown>).filter(([key]) => key !== "investigation" && key !== "jobScope"));
      const bodyRest = Object.fromEntries(Object.entries(body).filter(([key]) => key !== "investigation" && key !== "jobScope"));
      if (JSON.stringify(bodyRest) !== JSON.stringify(JSON.parse(JSON.stringify(rest)))) problems.push({ index, kind: "rest", bodyRest, rest });
    }
    console.log(`PINK_CONTROLLER sent=${sent} refused=${refused} problems=${problems.length}`);
    expect(sent).toBeGreaterThan(500);
    expect(refused).toBeGreaterThan(500);
    expect(problems.slice(0, 5)).toEqual([]);
  });

  it("M-4 declareAnalysis per mode × validation state; every declaration is accepted by the server rule as declared", () => {
    const scope = { jobPoolUploadId: UPLOAD, poolRoleId: ROLE };
    const validations: JobScopeValidation[] = [
      { state: "NONE" }, { state: "CHECKING", scope }, { state: "INVALID", scope, reason: "ROLE_NOT_IN_POOL" },
      { state: "VALID", scope, roleTitle: "Architect", organizationId: "org", organizationName: null, poolName: "PINK" }
    ];
    const table: string[] = [];
    for (const mode of INVESTIGATION_MODES) for (const validation of validations) {
      const declared = declareAnalysis(mode, validation);
      table.push(`${mode}/${validation.state}=${declared.ok ? declared.mode : declared.code}`);
      if (mode === "JOB") { expect(declared).toEqual({ ok: false, code: "ERR_INVESTIGATION_MODE_NOT_ANALYZABLE" }); continue; }
      if (mode === "PERSON") {
        expect(declared).toEqual({ ok: true, mode: "PERSON", declaration: { investigation: { mode: "PERSON" } } });
        expect(server({ documents: [], ...(declared.ok ? declared.declaration : {}) })).toEqual({ ok: true, mode: "PERSON", selection: null });
        continue;
      }
      if (validation.state !== "VALID") { expect(declared).toEqual({ ok: false, code: "ERR_JOB_SCOPE_REQUIRED" }); continue; }
      expect(declared).toEqual({ ok: true, mode: "PERSON_X_JOB", declaration: { investigation: { mode: "PERSON_X_JOB" }, jobScope: scope } });
      expect(server({ documents: [], ...(declared.ok ? declared.declaration : {}) })).toEqual({ ok: true, mode: "PERSON_X_JOB", selection: scope });
    }
    console.log(`PINK_DECLARE ${table.join(" ")}`);
  });

  it("M-5 only the three exact URL values are a choice; the ground state is PERSON_X_JOB", () => {
    expect(DEFAULT_INVESTIGATION_MODE).toBe("PERSON_X_JOB");
    for (const mode of INVESTIGATION_MODES) expect(investigationModeFromParam(mode)).toBe(mode);
    for (const value of [undefined, null, "", "person", "Person", "PERSON ", " PERSON", "PERSON\u0000", "PERSON×JOB", "PERSON_X_JOB,JOB", "JOB;PERSON", "PERSON%20", "0", "undefined", "null", "__proto__", "constructor", "toString"]) {
      expect(investigationModeFromParam(value as string | null | undefined)).toBeNull();
    }
  });
});
