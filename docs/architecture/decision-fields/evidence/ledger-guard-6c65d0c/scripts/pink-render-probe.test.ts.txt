  ["U+FE0F variation selector", "A go️od fit.", "A good fit."],

  ["U+2067 RTL isolate (reversed)", "⁧.tif doog A⁩", "A good fit. (rendered right to left)"],

it("render vs guard", () => {
  const o: Record<string, { guard: string; renders: string }> = {};
  for (const [name, text, renders] of cases) o[name] = { guard: containsClaim(text, name.startsWith("operand Go") ? ["Go"] : []) ? "REFUSED" : "ADMITTED", renders };
  writeFileSync(process.env.PINK_PROBE_OUT!, JSON.stringify(o, null, 1));
});
