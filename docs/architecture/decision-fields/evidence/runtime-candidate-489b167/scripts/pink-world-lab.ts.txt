// PINK runtime-candidate lab: a fresh marked disposable DB restored from a dump (source untouched), a production build of the
// checked-out commit, `next start` with PINK's net sentinel, NO worker, NO Gemini key. argv: <lab dir> <port> (--dump <file> | --reuse <state.json>) [--build] [--keep]
import { spawn, spawnSync } from "node:child_process";
import { mkdirSync, writeFileSync, readFileSync, unlinkSync, appendFileSync } from "node:fs";
import { resolve } from "node:path";
const [DIR, PORT] = process.argv.slice(2, 4); const arg = (k: string) => { const i = process.argv.indexOf(k); return i > 0 ? process.argv[i + 1] : undefined; };
const W = process.cwd(); mkdirSync(DIR, { recursive: true }); const STATE = resolve(DIR, "state.json");
const SENTINEL = process.env.PINK_SENTINEL!; const KEEP = process.argv.includes("--keep");
(async () => {
  const { createDisposableTestDatabase, dropDisposableTestDatabase, verifyDisposableTestDatabase } = await import(`${W}/lib/database-isolation/verification`);
  const { requireTestDatabaseAdminUrl } = await import(`${W}/scripts/test-db/admin-url`);
  let db: { url: string; databaseName: string };
  if (arg("--reuse")) { const st = JSON.parse(readFileSync(arg("--reuse")!, "utf8")); const u = new URL(requireTestDatabaseAdminUrl()); u.pathname = `/${st.database}`; db = await verifyDisposableTestDatabase(u.toString()); console.log(`[pink-lab] reuse verified ${db.databaseName}`); }
  else if (process.argv.includes("--seed-hr")) {
    db = await createDisposableTestDatabase(requireTestDatabaseAdminUrl()); console.log(`[pink-lab] created ${db.databaseName} (HR world)`);
    await verifyDisposableTestDatabase(db.url); process.env.DATABASE_URL = db.url;
    const postgres = (await import(`${W}/node_modules/postgres/src/index.js`)).default; const sql = postgres(db.url, { max: 2, onnotice: () => undefined });
    const { registerUnifiedPersistenceSchema } = await import(`${W}/lib/persistence/unified-schema-registration`); await registerUnifiedPersistenceSchema(sql);
    { const jp = await import(`${W}/lib/career/job-pool/persistence-schema`); await jp.registerJobPoolPersistenceSchema(sql); }
    const world = await import(`${W}/test/career/hr-decision-loop/fixtures/hr-loop-postgres-world`); const data = await world.seedHrLoopWorldInto(sql, db.url, db.databaseName);
    writeFileSync(resolve(DIR, "hr-ids.txt"), `${data.contextA} ${data.contextB}`); writeFileSync(resolve(DIR, "hr-child.txt"), String(data.g2?.childRevisionId ?? ""));
    await sql.end({ timeout: 5 }); console.log(`[pink-lab] seeded HR world`);
  }
  else {
    db = await createDisposableTestDatabase(requireTestDatabaseAdminUrl()); console.log(`[pink-lab] created ${db.databaseName}`);
    const r = spawnSync("pg_restore", ["--no-owner", "--no-privileges", `--dbname=${db.url}`, arg("--dump")!], { encoding: "utf8" });
    console.log(`[pink-lab] pg_restore exit ${r.status} stderr-lines ${(r.stderr ?? "").split("\n").filter(Boolean).length}`); if (r.stderr) appendFileSync(resolve(DIR, "restore.log"), r.stderr.replace(/postgres(ql)?:\/\/[^ ]*/g, "postgres://***"));
    await verifyDisposableTestDatabase(db.url);
  }
  const env = { ...process.env, DATABASE_URL: db.url, NEXT_TELEMETRY_DISABLED: "1", GEMINI_API_KEY: "", NODE_OPTIONS: `--require ${SENTINEL}`, PINK_NET_LOG: resolve(DIR, "net.log"), NODE_ENV: "production" } as NodeJS.ProcessEnv;
  delete env.GEMINI_MODEL; delete env.GOOGLE_GEMINI_BASE_URL;
  if (process.argv.includes("--build")) { const b = spawnSync(process.execPath, [resolve(W, "node_modules/next/dist/bin/next"), "build", "--webpack"], { cwd: W, env, stdio: ["ignore", "ignore", "pipe"], encoding: "utf8" }); console.log(`[pink-lab] next build exit ${b.status}`); if (b.status !== 0) { console.log((b.stderr ?? "").slice(-1500)); } }
  const server = spawn(process.execPath, [resolve(W, "node_modules/next/dist/bin/next"), "start", "-p", PORT, "-H", "127.0.0.1"], { cwd: W, env, stdio: ["ignore", "pipe", "pipe"] });
  server.stdout?.on("data", c => appendFileSync(resolve(DIR, "server.log"), String(c))); server.stderr?.on("data", c => appendFileSync(resolve(DIR, "server.log"), String(c)));
  const base = `http://127.0.0.1:${PORT}`; let up = false;
  for (let i = 0; i < 240 && !up; i += 1) { try { const r = await fetch(`${base}/api/career/job-pools`); up = r.status > 0; await r.text(); } catch { await new Promise(r => setTimeout(r, 500)); } }
  writeFileSync(STATE, JSON.stringify({ port: +PORT, base, database: db.databaseName, pid: process.pid, serverPid: server.pid, gemini: "STANDIN", worker: null }, null, 2), { mode: 0o600 });
  console.log(`PINK LAB READY ${base} database=${db.databaseName} server=${server.pid} up=${up}`);
  const stop = async () => { try { server.kill("SIGTERM"); } catch {} await new Promise(r => setTimeout(r, 3000)); if (server.exitCode === null) try { server.kill("SIGKILL"); } catch {}
    if (!KEEP) { await dropDisposableTestDatabase(db.url); console.log(`[pink-lab] dropped ${db.databaseName}`); } else console.log(`[pink-lab] kept ${db.databaseName}`);
    try { if (!KEEP) unlinkSync(STATE); } catch {} process.exit(0); };
  process.on("SIGTERM", stop); process.on("SIGINT", stop); server.on("exit", c => appendFileSync(resolve(DIR, "server.log"), `\n[server exit ${c}]\n`));
})().catch(e => { console.error(String(e).replace(/postgres(ql)?:\/\/[^ ]*/g, "postgres://***")); process.exit(1); });
