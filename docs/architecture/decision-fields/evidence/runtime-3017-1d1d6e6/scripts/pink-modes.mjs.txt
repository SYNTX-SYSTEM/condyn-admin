const snap = async () => p.evaluate(() => { const q = s => document.querySelector(s); const ids = [...new Set([...document.querySelectorAll("[data-testid]")].map(e => e.dataset.testid.replace(/[0-9A-F]{8,}|ANL_\d+_\d+|role_[a-z_]+/g, "#")))].sort(); const text = document.body?.innerText ?? "";
