// PINK: evaluates the committed focusNameSize code (extracted verbatim from SyntxField.tsx of each commit, types stripped) for every
// fixture pool name and cosmos-centre label. argv: <dir with sf-<commit>.tsx and the fixture JSONs> <commit...>
import { readFileSync } from "node:fs";
const [DIR, ...COMMITS] = process.argv.slice(2);
const load = c => {
  const src = readFileSync(`${DIR}/sf-${c}.tsx`, "utf8");
  const m = src.match(/const POOL_NAME_STEPS[\s\S]*?\n(?=\/\*\* The organization cosmos' centre)/);
  if (!m) return null;
  const js = m[0].replace(/: ReadonlyArray<readonly \[[^\]]*\]>/g, "").replace(/: string/g, "");
  return new Function(`${js}\nreturn { pool: n => focusNameSize(n, POOL_NAME_STEPS, "10.5px"), centre: l => focusNameSize(l, CENTRE_NAME_STEPS, "11px") };`)();
};
const fns = Object.fromEntries(COMMITS.map(c => [c, load(c) ?? { pool: () => "10.5px", centre: () => "11px" }]));
let n = 0, diff = 0;
for (const f of ["DLL_JOBPOOL_AI_DATA", "DLL_JOBPOOL_ENGINEERING", "DLL_JOBPOOL_ENTERPRISE_SCALE"]) {
  const j = JSON.parse(readFileSync(`${DIR}/${f}.json`, "utf8"));
  const rows = [["pool", j.pool.name, c => fns[c].pool(j.pool.name)], ...j.organizations.map((o, i) => { const l = `${String(i + 1).padStart(2, "0")} ${o.name}`; return ["centre", l, c => fns[c].centre(l)]; })];
  for (const [kind, label, get] of rows) { n += 1; const v = COMMITS.map(get); const d = v[v.length - 1] !== v[v.length - 2]; if (d) diff += 1;
    const word = Math.max(...label.toUpperCase().split(/\s+/).map(p => p.length));
    console.log(`${f.replace("DLL_JOBPOOL_", "")} ${kind.padEnd(6)} len ${String(label.toUpperCase().length).padStart(2)} word ${String(word).padStart(2)} ${COMMITS.map((c, i) => `${c} ${v[i]}`).join(" | ")}${d ? "  CHANGED" : ""} · ${label}`); }
}
console.log(`names ${n}; size differs between ${COMMITS.at(-2)} and ${COMMITS.at(-1)}: ${diff}`);
