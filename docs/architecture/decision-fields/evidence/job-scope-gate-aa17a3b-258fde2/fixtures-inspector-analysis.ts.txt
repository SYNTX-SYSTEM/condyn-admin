import { db } from "../../../lib/career/db/client";
import { careerAnalyses } from "../../../lib/career/db/schema";
import {
  GeminiCapabilityDiscoveryProvider, PostgresCandidateSourceBundleRepository, PostgresCapabilityCoreRepository,
  PostgresCapabilityProposalProjectionReferenceRepository, computeSourceBundleHash, createSourceDocument, runCapabilityConvergence,
  runCapabilityDiscovery, type CapabilityDiscoveryRun
} from "../../../lib/career/capability-core";
import { JobRepository } from "../../../lib/career/orchestration/job-repository";
import { createJob } from "../../../lib/career/orchestration/job";
import { renderInvestigationScope, type ResolvedJobScope } from "../../../lib/career/job-scope";

export const INSPECTOR_QUOTE = "Led the migration of the dispatch platform to event-driven services.";
const cv = createSourceDocument({ docId: "CV", title: "cv.synthetic.pdf", rawContent: INSPECTOR_QUOTE });
export const INSPECTOR_AUDIT = { source_documents_examined: 1, capability_count: 1, atomic_capability_count: 1, composite_capability_count: 0, attribution_pass_completed: true, target_state_ownership_pass_completed: true, atomic_extraction_pass_completed: true, method_capability_pass_completed: true, composite_reconstruction_pass_completed: true, global_convergence_pass_completed: true, inventory_reconciliation_pass_completed: true, final_reconciliation_produced_new_capabilities: false, unresolved_target_operations: 0, segments_classified_as_external_source_content: 0, segments_classified_as_target_subject_operation: 0, segments_classified_as_target_subject_designed_target_state: 0, segments_classified_as_target_organization_capability: 0, segments_excluded_due_to_attribution_ambiguity: 0 };

/**
 * One scoped analysis the way the worker leaves it (PINK's own seed, disposable database): real Discovery behind the real Gemini
 * provider with a fake client returning `raw`, real Convergence, an enqueued job row with a document body (the caller deletes it), the source bundle,
 * the analysis row and the projection reference. `onGeminiRequest` counts provider requests.
 */
export async function seedScopedAnalysis(scope: ResolvedJobScope, documentBody: string, raw: string, onGeminiRequest: () => void): Promise<{ analysisId: string; jobId: string; run: CapabilityDiscoveryRun }> {
  const capabilityRepository = new PostgresCapabilityCoreRepository(db);
  const client = { models: { async generateContent() { onGeminiRequest(); return { text: raw, candidates: [{ finishReason: "STOP" }], modelVersion: "pink-fake", responseId: "pink-pg", usageMetadata: {} }; } } };
  const discovered = await runCapabilityDiscovery([cv], { kernelVersion: "discovery-v1", schemaVersion: "discovery-schema-v1" }, { kernelResolver: { resolve: async () => ({ kernelVersion: "discovery-v1", templateId: "T", versionId: "V", checksum: "C", plainTextContent: "PINK_KERNEL" }) }, provider: new GeminiCapabilityDiscoveryProvider({ model: "pink-fake", client }), repository: capabilityRepository, investigationScope: renderInvestigationScope(scope) });
  const run = (discovered as { run: CapabilityDiscoveryRun }).run;
  const convergence = await runCapabilityConvergence((await capabilityRepository.getRunById(run.runId))!, { kernelVersion: "convergence-v1", schemaVersion: "convergence-schema-v1", algorithmVersion: "algorithm-v1" }, { kernelResolver: { resolve: async () => ({ kernelVersion: "convergence-v1", templateId: "T", versionId: "V", checksum: "CC", plainTextContent: "PINK_CONVERGENCE" }) }, provider: { providerName: "pink-spy", model: "m", execute: async (request: { userPrompt: string }) => {
    const ids = (JSON.parse(request.userPrompt.slice(request.userPrompt.indexOf("["))) as Array<{ candidate_id: string }>).map((item) => item.candidate_id);
    return { convergenceOutput: { convergence_version: "convergence-v1", groups: [{ group_key: "g", member_candidate_ids: ids, canonical_name: "Event-driven migration", capability_scope: "ATOMIC" as const, structural_definition: "d", primary_domain: null }], relations: [], reconciliation_audit: { input_candidate_count: ids.length, grouped_candidate_count: ids.length, group_count: 1, same_capability_merge_count: ids.length - 1, unresolved_relation_count: 0, reconciliation_pass_completed: true } } };
  } }, repository: capabilityRepository });
  const job = createJob("CAREER_ANALYSIS", { sourceType: "BATCH", sourceData: { documents: [{ type: "pdf", name: "cv.synthetic.pdf", content: documentBody }], jobScope: scope.selection } } as never);
  await new JobRepository(db).enqueueJob(job);
  const analysisId = job.jobId.replace("JOB_", "ANL_");
  await new PostgresCandidateSourceBundleRepository(db).persistCandidateSourceBundle({ candidateSourceBundleId: `CSB_${job.jobId}`, sourceBundleHash: computeSourceBundleHash([cv]), documents: [cv], schemaVersion: "CANDIDATE_SOURCE_BUNDLE_V1", createdAt: "2026-10-15T10:00:00.000Z" } as never);
  await db.insert(careerAnalyses).values({ analysisId, createdAt: convergence.run.completedAt, validationState: "VERIFIED", payload: {} } as never);
  await new PostgresCapabilityProposalProjectionReferenceRepository(db).save({ analysisId, jobId: job.jobId, discoveryRunId: run.runId, convergenceRunId: convergence.run.convergenceRunId, candidateSourceBundleId: `CSB_${job.jobId}`, sourceBundleHash: run.sourceBundleHash, createdAt: convergence.run.completedAt });
  return { analysisId, jobId: job.jobId, run };
}
