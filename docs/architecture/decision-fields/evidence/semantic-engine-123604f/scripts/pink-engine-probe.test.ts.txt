it("one attempt", async () => {
  const P = proposals.slice(0, 2), R: EngineRequirement[] = [{ key: { jobPoolUploadId: "U", poolRoleId: "r", poolRequirementId: "q1" }, capability_name: "Python", required_level: "L3", domain: "", necessity: "REQUIRED" }];
  const mk = (reserveOk = true) => { const log: string[] = []; return { log, ledger: { reserve: async () => { log.push("reserve"); return reserveOk; }, consume: async () => { log.push("consume"); }, release: async () => { log.push("release"); } }, writer: { saveIfAbsent: async (r: any[]) => { log.push(`save ${r.length}`); return { inserted: r.length, duplicates: 0, divergent: [] }; } } }; };
  const run = async (provider: any, reserveOk = true, props = P, reqs = R) => { const m = mk(reserveOk); let calls = 0; const prov = { evaluate: async (q: any) => { calls += 1; m.log.push("call"); return provider(q); } }; const r = await evaluateRole({ engine: STAND_IN, provider: prov, ledger: m.ledger, writer: m.writer, proposals: props, pendingRequirements: reqs, now: () => "2026-10-29T00:00:00.000Z" }).catch(e => ({ thrown: String(e).slice(0, 80) })); return { outcome: (r as any).status ?? r, code: (r as any).code, calls, log: m.log.join(" → ") }; };
  out.attempt = {
    success: await run((q: any) => new StandInSemanticProvider().evaluate(q)),
    providerThrows: await run(() => { throw new Error("ERR_PROVIDER_TIMEOUT: x"); }),
    providerThrowsPlain: await run(() => { throw new Error("socket hang up"); }),
    refusedOutput: await run(() => ({ rawText: "not json" })),
    partlyRefused: await run(async (q: any) => { const t = JSON.parse((await new StandInSemanticProvider().evaluate(q)).rawText); t.relations[0].relationExplanation = "score 90"; return { rawText: JSON.stringify(t) }; }),
    noBudget: await run((q: any) => new StandInSemanticProvider().evaluate(q), false),
    emptyRequirements: await run((q: any) => new StandInSemanticProvider().evaluate(q), true, P, []),
    emptyProposals: await run((q: any) => new StandInSemanticProvider().evaluate(q), true, [], R)
  };
  // consume or writer failing after the call
  const m = mk(); m.ledger.consume = async () => { m.log.push("consume-throws"); throw new Error("ledger down"); };
  out.attempt.consumeThrows = await evaluateRole({ engine: STAND_IN, provider: new StandInSemanticProvider(), ledger: m.ledger, writer: m.writer, proposals: P, pendingRequirements: R, now: () => "t" }).then(r => r.status, e => `thrown: ${String(e).slice(0, 40)} · log ${m.log.join(" → ")}`);
  const m2 = mk(); m2.writer.saveIfAbsent = async () => { m2.log.push("save-throws"); throw new Error("ERR_RELATION_CACHE_RECORD_CORRUPT:X"); };
  out.attempt.writerThrows = await evaluateRole({ engine: STAND_IN, provider: new StandInSemanticProvider(), ledger: m2.ledger, writer: m2.writer, proposals: P, pendingRequirements: R, now: () => "t" }).then(r => r.status, e => `thrown: ${String(e).slice(0, 40)} · log ${m2.log.join(" → ")}`);
});
