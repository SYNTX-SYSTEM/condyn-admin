// PINK §10: role-card box form (orbit / packed / compact) and domain line per viewport at the organisation and role level. GET only.
// argv: <base> <out dir>
import { pathToFileURL } from "node:url";
const [BASE, OUT] = process.argv.slice(2);
const pw = await import(pathToFileURL(process.env.CONDYN_PLAYWRIGHT_MODULE).href); const { chromium } = pw.chromium ? pw : pw.default;
const b = await chromium.launch(); let blocked = 0; const out = [];
const DLL = "JPOOL_B94850C439974E27496FE3B1A9A076B5";
for (const [n, w, h, d] of [["1920x1080", 1920, 1080, 1], ["1600x900", 1600, 900, 1], ["1440x900", 1440, 900, 1], ["1280x720", 1280, 720, 1], ["zoom200-1920x1080", 960, 540, 2], ["zoom200-1280x720", 640, 360, 2]]) {
  const ctx = await b.newContext({ viewport: { width: w, height: h }, deviceScaleFactor: d }); await ctx.addInitScript(() => { try { localStorage.setItem("condyn_onboarding_seen", "true"); } catch {} });
  await ctx.route("**/*", r => { const m = r.request().method(); if (m !== "GET" && m !== "HEAD") { blocked += 1; return r.abort(); } return r.continue(); });
  const p = await ctx.newPage();
  for (const [scene, q] of [["org-PXJ", `investigationMode=PERSON_X_JOB&jobPoolUploadId=${DLL}&jobOrganizationId=org_kestrel`], ["role-PXJ", `investigationMode=PERSON_X_JOB&jobPoolUploadId=${DLL}&jobOrganizationId=org_kestrel&jobRoleId=role_kestrel_fullstack`], ["org-PERSON", `investigationMode=PERSON&jobPoolUploadId=${DLL}&jobOrganizationId=org_kestrel`]]) {
    await p.goto(`${BASE}/career/demo?${q}`, { waitUntil: "domcontentloaded" }); await new Promise(r => setTimeout(r, 6000));
    const cards = await p.evaluate(() => [...document.querySelectorAll("[data-card-box]")].map(e => ({ role: (e.closest('[data-testid^="syntx-role-"]')?.dataset.testid ?? e.dataset.testid ?? "").replace("syntx-role-role_kestrel_", ""), box: e.dataset.cardBox, domain: !!e.querySelector("[data-sx-role-domain]") })));
    out.push({ vp: n, scene, cards: cards.map(c => `${c.role}:${c.box}${c.domain ? "+domain" : ""}`).join(" ") });
    if (n === "1280x720" && scene === "org-PXJ") await p.screenshot({ path: `${OUT}/V18-role-cards-org-1280x720.jpg`, type: "jpeg", quality: 75 });
    if (n === "1920x1080" && scene === "org-PXJ") await p.screenshot({ path: `${OUT}/V18-role-cards-org-1920x1080.jpg`, type: "jpeg", quality: 75 });
  }
  await ctx.close();
}
console.log(JSON.stringify({ blocked, out }, null, 1)); await b.close();
