// PINK §10 reach probe (V-01 … V-03, V-14, V-16, O-PXJ-720): every target at six viewports — at load, after a user's own scrolling
// (only overflow auto/scroll ancestors and the page), 5-point elementFromPoint, Playwright trial click (actionability, no action).
// B2 rows by mouse wheel (60 px and 240 px steps): a row counts when it was fully in view AND its centre hit itself.
// argv: <base> <out dir> <real|standin> [analysisId]. real: GET only — every non-GET is aborted and counted. Quotes never recorded.
import { writeFileSync, mkdirSync } from "node:fs";
import { pathToFileURL } from "node:url";
const [BASE, OUT, KIND = "real", ANL_ARG = ""] = process.argv.slice(2); mkdirSync(OUT, { recursive: true });
const pw = await import(pathToFileURL(process.env.CONDYN_PLAYWRIGHT_MODULE).href); const { chromium } = pw.chromium ? pw : pw.default;
const DLL = "JPOOL_B94850C439974E27496FE3B1A9A076B5", KES = "org_kestrel", ROLE = "role_kestrel_fullstack";
const ANL = ANL_ARG || "ANL_1792628993472_10";
const sleep = ms => new Promise(r => setTimeout(r, ms));
const VPS = [["1920x1080", 1920, 1080, 1], ["1600x900", 1600, 900, 1], ["1440x900", 1440, 900, 1], ["1280x720", 1280, 720, 1], ["zoom200-1920x1080", 960, 540, 2], ["zoom200-1280x720", 640, 360, 2]];
const sel = `&jobScopePoolUploadId=${DLL}&jobScopeRoleId=${ROLE}`;
const SCENES = [
  ["S1-start", `${BASE}/career/demo`, ["job-pool-select-" + DLL, "syntx-pools-toggle", "semantic-zoom-telemetry", "job-scope-indicator", "investigation-mode-PERSON", "investigation-mode-PERSON_X_JOB", "investigation-mode-JOB"]],
  ["S2-role-view", `${BASE}/career/demo?jobPoolUploadId=${DLL}&jobOrganizationId=${KES}&jobRoleId=${ROLE}`, ["syntx-job-scope-select", "syntx-path-FIELD", "syntx-path-ORGANIZATION", "syntx-return-btn", "syntx-pools-toggle", "job-pool-panel-toggle"]],
  ["S2b-role-view-selected", `${BASE}/career/demo?investigationMode=PERSON_X_JOB&jobPoolUploadId=${DLL}&jobOrganizationId=${KES}&jobRoleId=${ROLE}${sel}`, ["analysis-picker-toggle", "syntx-job-scope-clear", "field-focus-capability-btn", "syntx-pools-toggle"]],
  ["S3-capability-no-analysis", `${BASE}/career/demo?investigationMode=PERSON_X_JOB&jobPoolUploadId=${DLL}${sel}&focus=CAPABILITY`, ["add-text-source-btn", "add-pdf-source-btn", "analysis-picker-toggle", "investigation-mode-PERSON", "investigation-mode-PERSON_X_JOB", "investigation-mode-JOB", "job-pool-panel-toggle", "field-focus-capability-btn", "semantic-zoom-telemetry"]],
  ["S4-b2-available", `${BASE}/career/demo?investigationMode=PERSON_X_JOB&jobPoolUploadId=${DLL}${sel}&analysisId=${ANL}&focus=CAPABILITY`, ["add-text-source-btn", "add-pdf-source-btn", "investigation-mode-JOB", "job-pool-panel-toggle", "semantic-zoom-telemetry", "pxj-first-counts", "pxj-first-state", "pxj-first-coverage-boundary"]],
  ["S6-gate-closed", `${BASE}/career/demo?investigationMode=PERSON_X_JOB&jobPoolUploadId=${DLL}`, ["field-focus-capability-btn", "capability-gate-reason", "capability-gate-cause", "capability-gate-way"]]
];
const rec = { kind: KIND, evidenceClass: KIND === "real" ? "PERSISTED_REAL_RESULT" : "STAND_IN", note: KIND === "real" ? "GELB :41872, GET only" : "STAND-IN · keine Modellausführung", measures: [], wheel: [], blocked: 0, pageErrors: [] };
const browser = await chromium.launch();
for (const [vpName, w, h, dsf] of VPS) {
  const ctx = await browser.newContext({ viewport: { width: w, height: h }, deviceScaleFactor: dsf });
  await ctx.addInitScript(() => { try { localStorage.setItem("condyn_onboarding_seen", "true"); } catch {} });
  if (KIND === "real") await ctx.route("**/*", r => { const m = r.request().method(); if (m !== "GET" && m !== "HEAD") { rec.blocked += 1; return r.abort(); } return r.continue(); });
  const page = await ctx.newPage(); page.on("pageerror", e => rec.pageErrors.push(`${vpName} ${String(e).slice(0, 140)}`));
  for (const [scene, url, targets] of SCENES) {
    await page.goto(url, { waitUntil: "domcontentloaded" }); await sleep(6000);
    const geo = id => page.evaluate(id => { const e = document.querySelector(`[data-testid="${id}"]`) ?? document.querySelector(`[data-testid^="${id}"]`); if (!e) return null; const r = e.getBoundingClientRect(); const vw = innerWidth, vh = innerHeight;
      const pts = [[r.left + r.width / 2, r.top + r.height / 2], [r.left + 3, r.top + 3], [r.right - 3, r.top + 3], [r.left + 3, r.bottom - 3], [r.right - 3, r.bottom - 3]];
      let inView = 0, hit = 0; const cover = new Set(); for (const [x, y] of pts) { if (x < 0 || y < 0 || x >= vw || y >= vh) continue; inView += 1; const t = document.elementFromPoint(x, y); if (t && (t === e || e.contains(t))) hit += 1; else if (t) cover.add(t.closest("[data-testid]")?.dataset.testid ?? t.tagName.toLowerCase()); }
      const full = r.top >= 0 && r.left >= 0 && r.bottom <= vh && r.right <= vw; return { y: Math.round(r.top), h: Math.round(r.height), w: Math.round(r.width), full, hit: `${hit}/${inView}`, cover: [...cover].slice(0, 3), display: getComputedStyle(e).visibility };
    }, id);
    for (const id of targets) {
      const atLoad = await geo(id); if (!atLoad) { rec.measures.push({ vp: vpName, scene, id, present: false }); continue; }
      const loc = page.locator(`[data-testid="${id}"]`).first();
      // which ancestors a scrollIntoView would move, and whether a user can scroll them (overflow auto/scroll, or the page itself)
      const before = await page.evaluate(id => { const e = document.querySelector(`[data-testid="${id}"]`); const out = []; for (let a = e?.parentElement; a; a = a.parentElement) { if (a.scrollHeight > a.clientHeight + 1 || a.scrollWidth > a.clientWidth + 1) out.push({ tid: a.dataset?.testid ?? a.tagName.toLowerCase(), top: a.scrollTop, left: a.scrollLeft, oy: getComputedStyle(a).overflowY, ox: getComputedStyle(a).overflowX, page: a === document.scrollingElement }); } return out; }, id);
      await loc.scrollIntoViewIfNeeded({ timeout: 4000 }).catch(() => {}); await sleep(400);
      const moved = await page.evaluate(([id, before]) => { const e = document.querySelector(`[data-testid="${id}"]`); const out = []; let i = 0; for (let a = e?.parentElement; a; a = a.parentElement) { if (a.scrollHeight > a.clientHeight + 1 || a.scrollWidth > a.clientWidth + 1) { const b = before.find(x => x.tid === (a.dataset?.testid ?? a.tagName.toLowerCase())); if (b && (b.top !== a.scrollTop || b.left !== a.scrollLeft)) out.push({ tid: b.tid, userScrollable: b.page || /auto|scroll/.test(b.oy) || /auto|scroll/.test(b.ox), oy: b.oy }); i += 1; } } return out; }, [id, before]);
      const afterScroll = await geo(id);
      let trial = "ok"; try { await loc.click({ trial: true, timeout: 2500 }); } catch (e) { trial = String(e).split("\n")[0].slice(0, 90); }
      const userReach = moved.every(m => m.userScrollable);
      rec.measures.push({ vp: vpName, scene, id, present: true, atLoad, afterScroll, scrolled: moved, userReach, trial });
      await page.evaluate(() => { for (const a of document.querySelectorAll("*")) if (a.scrollTop) a.scrollTop = 0; scrollTo(0, 0); }); await sleep(250);
    }
    if (scene === "S4-b2-available") {
      // head, binding and counts at once (at load), then rows by wheel
      const trio = await page.evaluate(() => { const vis = s => { const e = document.querySelector(s); if (!e) return null; const r = e.getBoundingClientRect(); return r.top >= 0 && r.bottom <= innerHeight && r.left >= 0 && r.right <= innerWidth; }; const head = document.querySelector('[data-testid="pxj-first-screen"]')?.firstElementChild; const hv = head ? (r => r.top >= 0 && r.bottom <= innerHeight)(head.getBoundingClientRect()) : null; return { head: hv, boundary: vis('[data-testid="pxj-first-coverage-boundary"]'), binding: vis('[data-pxj-section="binding"]'), counts: vis('[data-testid="pxj-first-counts"]'), status: vis('[data-testid="pxj-first-state"]') }; });
      for (const step of [60, 240]) {
        await page.goto(url, { waitUntil: "domcontentloaded" }); await sleep(6000);
        const rows = await page.evaluate(() => [...document.querySelectorAll('[data-testid^="pxj-first-req-"]')].map(e => e.dataset.testid));
        const panel = await page.evaluate(() => { const r = document.querySelector('[data-testid="pxj-first-screen"]')?.getBoundingClientRect(); if (!r) return null; return { x: Math.max(4, Math.min(innerWidth - 4, r.left + r.width / 2)), y: Math.max(4, Math.min(innerHeight - 4, (Math.max(r.top, 0) + Math.min(r.bottom, innerHeight)) / 2)) }; });
        const seen = Object.fromEntries(rows.map(r => [r, { full: false, hit: false, both: false, maxShare: 0 }]));
        if (panel) await page.mouse.move(panel.x, panel.y);
        for (let k = 0; k < 140; k += 1) {
          const st = await page.evaluate(ids => ids.map(id => { const e = document.querySelector(`[data-testid="${id}"]`); const r = e.getBoundingClientRect(); const vt = Math.max(r.top, 0), vb = Math.min(r.bottom, innerHeight); let clipTop = 0, clipBottom = innerHeight; for (let a = e.parentElement; a; a = a.parentElement) { const cs = getComputedStyle(a); if (/auto|scroll|hidden|clip/.test(cs.overflowY)) { const ar = a.getBoundingClientRect(); clipTop = Math.max(clipTop, ar.top); clipBottom = Math.min(clipBottom, ar.bottom); } } const t = Math.max(vt, clipTop), b = Math.min(vb, clipBottom); const share = Math.max(0, b - t) / r.height; const cy = (t + b) / 2, cx = r.left + r.width / 2; const hitEl = share > 0 ? document.elementFromPoint(cx, cy) : null; return { id, full: share >= 0.999, share, hit: !!hitEl && e.contains(hitEl), h: Math.round(r.height), clip: Math.round(clipBottom - clipTop) }; }), rows);
          for (const s of st) { const o = seen[s.id]; o.full ||= s.full; o.hit ||= s.hit; o.both ||= s.full && s.hit; o.maxShare = Math.max(o.maxShare, Math.round(s.share * 100) / 100); o.h = s.h; o.clip = s.clip; }
          if (rows.length && Object.values(seen).every(o => o.both)) break;
          await page.mouse.wheel(0, step); await sleep(120);
          if (panel) { const p2 = await page.evaluate(() => { const r = document.querySelector('[data-testid="pxj-first-screen"]')?.getBoundingClientRect(); return r ? { x: Math.max(4, Math.min(innerWidth - 4, r.left + r.width / 2)), y: Math.max(4, Math.min(innerHeight - 4, (Math.max(r.top, 0) + Math.min(r.bottom, innerHeight)) / 2)) } : null; }); if (p2) await page.mouse.move(p2.x, p2.y); }
        }
        rec.wheel.push({ vp: vpName, step, trio, rows: rows.length, fullAndHit: Object.values(seen).filter(o => o.both).length, perRow: Object.fromEntries(Object.entries(seen).map(([k, v]) => [k.replace("pxj-first-req-", ""), `${v.both ? "OK" : "NO"} share${v.maxShare} h${v.h} clip${v.clip}`])) });
      }
    }
  }
  await ctx.close(); console.log("VP", vpName, "measures", rec.measures.filter(m => m.vp === vpName).length);
}
writeFileSync(`${OUT}/pink-reach-${KIND}.json`, JSON.stringify(rec, null, 1));
const bad = rec.measures.filter(m => m.present && (!(m.afterScroll?.full) || !/^5\/5$/.test(m.afterScroll?.hit ?? "") || !m.userReach || m.trial !== "ok") && !/telemetry|gate-(reason|cause|way)|counts|pxj-first-state|coverage-boundary/.test(m.id));
console.log("DONE", JSON.stringify({ measures: rec.measures.length, notReachable: bad.map(m => `${m.vp}/${m.scene}/${m.id}:${m.afterScroll?.hit}:${m.afterScroll?.full}:${m.userReach}:${m.trial.slice(0, 40)}`), wheel: rec.wheel.map(x => `${x.vp}@${x.step}:${x.fullAndHit}/${x.rows}`), blocked: rec.blocked, errors: rec.pageErrors.length }));
await browser.close();
