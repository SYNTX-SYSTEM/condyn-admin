#!/bin/bash
# PINK G1: one fresh STAND-IN lab per commit in PINK's worktree (own PID, port 41873, disposable DB), PINK probe, stop.
G=<pink-scratch>; P=<local>
C=$1; L=$2; shift 2
cd $P && git checkout -q --detach $C && echo "HEAD $(git rev-parse --short HEAD) ($L)"
set -a; . ~/.config/condyn/test-db-role.env; set +a
unset DATABASE_URL CONDYN_ALLOW_SHARED_DATABASE GEMINI_API_KEY GEMINI_MODEL GOOGLE_GEMINI_BASE_URL VIDEO_LAB_GEMINI VIDEO_LAB_LIVE_GEMINI_GO VIDEO_LAB_GEMINI_ENV
export CONDYN_PLAYWRIGHT_MODULE=<local>
OUT=$G/out-$L; mkdir -p $OUT; LAB=$G/lab-$L; rm -rf "${LAB:?}"; mkdir -p $LAB
VIDEO_LAB_PORT=41873 VIDEO_LAB_DIR=$LAB npx tsx docs/career_analysis/evidence/video-readiness/scripts/video-lab.ts > $LAB/lab.log 2>&1 &
for k in $(seq 1 400); do grep -qE "VIDEO LAB READY|Error|failed" $LAB/lab.log && break; sleep 3; done
if grep -qE "VIDEO LAB READY" $LAB/lab.log; then
  grep -oE "gemini=[A-Z]+" $LAB/lab.log | head -1
  for VP in "$@"; do timeout 3600 node $G/pink/pink-g1-walk.mjs $LAB/state.json $P $OUT $L ${VP%x*} ${VP#*x} > $OUT/walk-$VP.log 2>&1; echo "walk $VP exit $? :: $(tail -1 $OUT/walk-$VP.log | cut -c1-900)"; done
  cp $LAB/net.jsonl $OUT/net.jsonl 2>/dev/null
else echo "lab failed"; grep -E "Error|failed" $LAB/lab.log | sed -E 's/(postgres[a-z]*:\/\/)[^ ]*/\1***/g' | head -5; fi
bash docs/career_analysis/evidence/video-readiness/scripts/stop-lab.sh $LAB; wait
git status --short | head -5; git checkout -q validation/hr-decision-loop-pink && echo "back on $(git branch --show-current)"
echo "LAB-END $L"
