  const fieldTextNoPanel = (() => { if (!root) return ""; const c = root.cloneNode(true); c.querySelectorAll('[data-testid="pxj-first-screen"],[data-pxj-quote]').forEach(n => n.remove()); return c.innerText.replace(/\s+/g, " ").trim(); })();

    coverage: q('[data-testid="pxj-first-coverage"]') ? attrs(q('[data-testid="pxj-first-coverage"]')) : null,
    boundary: q('[data-testid="pxj-first-coverage-boundary"]')?.innerText.replace(/\s+/g, " ") ?? null,
    counts: q('[data-testid="pxj-first-counts"]')?.innerText.replace(/\s+/g, " ") ?? null,
    rows: qa('[data-testid^="pxj-first-req-"]').map(attrs),

    proposals: qa('[data-testid^="pxj-first-proposal-"]').map(e => e.dataset.testid.replace("pxj-first-proposal-", "")),
    stages: qa('[data-testid^="focus-transition-stage-shell-"]').map(e => e.dataset.testid.slice(-2)),
    coupling: !!q('[data-testid="inter-orbital-coupling"]'), viewMatches: qa('[data-testid="hud-action-view-matches"]').length,

    disclosure: q('[data-testid="syntx-legacy-disclosure"]')?.innerText.replace(/\s+/g, " ") ?? null,
    hashPlaceholder: document.body.innerText.includes("HASH-"), unknownDoc: document.body.innerText.includes("Unknown Document"),
    fieldTextNoPanel, testids: [...new Set(qa("[data-testid]").map(e => e.dataset.testid))].sort()
  };
