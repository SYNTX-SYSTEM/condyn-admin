// PINK's own keyboard and focus-transition probe for frontend/focus-readability (9fb1df6, base 38ebc31), identical on both commits.
// Per field state (URL, requirement level via jobRequirementId): the Tab order (test id, aria label, box, in view, focus ring), and
// a keyboard descent FIELD → organization → role → requirement by Tab + Enter, then Escape steps, recording the observation kind,
// the URL and the focused element after each step. Stand-in lab only, GET pages only. argv: <lab state.json> <out.json>.
import { readFileSync, writeFileSync } from "node:fs";
import { pathToFileURL } from "node:url";
const [STATE, OUT] = process.argv.slice(2);
const pw = await import(pathToFileURL(process.env.CONDYN_PLAYWRIGHT_MODULE).href); const { chromium } = pw.chromium ? pw : pw.default;
const BASE = JSON.parse(readFileSync(STATE, "utf8")).base;
const ENG = JSON.parse(readFileSync("docs/examples/job-pool-tests/DLL_JOBPOOL_ENGINEERING.json", "utf8"));
const up = await (await fetch(`${BASE}/api/career/job-pools`, { method: "POST", headers: { "content-type": "application/json", "x-condyn-principal-actor-id": "PINK_FOCUS_LAB" }, body: JSON.stringify(ENG) })).json();
const P = up.jobPoolUploadId ?? up.upload?.jobPoolUploadId;
const res = { pool: P, tab: [], descent: [], errors: [] };
const browser = await chromium.launch(); let ctx, page;
const fresh = async (w, h) => { if (ctx) await ctx.close(); ctx = await browser.newContext({ viewport: { width: w, height: h }, reducedMotion: "reduce" }); await ctx.addInitScript(() => { try { localStorage.setItem("condyn_onboarding_seen", "1"); } catch {} }); page = await ctx.newPage(); page.on("pageerror", e => res.errors.push(String(e).slice(0, 200))); };
const go = async q => { await page.goto(`${BASE}/career/demo?focus=FIELD&jobPoolUploadId=${P}${q}&investigationMode=PERSON`, { waitUntil: "networkidle" }); await page.getByTestId("syntx-field").waitFor({ timeout: 60000 }); await page.waitForTimeout(1200); };
const active = () => page.evaluate(() => {
  const a = document.activeElement; if (!a || a === document.body) return { id: "body" };
  const host = a.closest("[data-testid]"); const b = a.getBoundingClientRect(); const s = getComputedStyle(a);
  const ring = s.outlineStyle !== "none" && parseFloat(s.outlineWidth) > 0 ? `outline ${s.outlineWidth}` : s.boxShadow !== "none" ? "shadow" : "none";
  const field = document.querySelector('[data-testid="syntx-field"]');
  return { id: (host && host.getAttribute("data-testid")) || a.tagName.toLowerCase(), own: a.getAttribute("data-testid"), aria: (a.getAttribute("aria-label") || a.textContent || "").trim().slice(0, 50), box: [Math.round(b.left), Math.round(b.top), Math.round(b.width), Math.round(b.height)], inView: b.top >= 0 && b.left >= 0 && b.bottom <= innerHeight && b.right <= innerWidth, ring, kind: field?.getAttribute("data-observation-kind"), scrollY: Math.round(scrollY) };
});
const kind = () => page.evaluate(() => ({ kind: document.querySelector('[data-testid="syntx-field"]')?.getAttribute("data-observation-kind"), url: location.search.replace(/jobPoolUploadId=[^&]+/, "jobPoolUploadId=P") }));
const step = async (name, fn) => { try { await fn(); } catch (e) { res.errors.push(`${name}: ${String(e).split("\n")[0].slice(0, 200)}`); console.log("FAIL", name, String(e).split("\n")[0].slice(0, 160)); } };
const STATES = [["FIELD", ""], ["ORGANIZATION", "&jobOrganizationId=org_kestrel"], ["ROLE", "&jobOrganizationId=org_kestrel&jobRoleId=role_kestrel_fullstack"],
  ["REQUIREMENT", "&jobOrganizationId=org_kestrel&jobRoleId=role_kestrel_fullstack&jobRequirementId=req_kestrel_fullstack_01"]];
for (const [W, H] of [[1920, 1080], [1280, 720]]) {
  await fresh(W, H);
  for (const [name, q] of STATES) await step(`tab ${name} ${W}`, async () => {
    await go(q); const seq = []; await page.evaluate(() => { document.activeElement?.blur?.(); window.focus(); }); // start of the document's tab order without a click (a click on the field's ground returns one level)
    for (let i = 0; i < 70; i += 1) { await page.keyboard.press("Tab"); seq.push(await active()); }
    res.tab.push({ state: name, w: W, start: await kind(), seq });
  });
  await step(`descent ${W}`, async () => {
    await go(""); await page.evaluate(() => { document.activeElement?.blur?.(); window.focus(); }); const log = [];
    const tabTo = async re => { for (let i = 0; i < 80; i += 1) { await page.keyboard.press("Tab"); const a = await active(); if (re.test(a.own || a.id)) return a; } return null; };
    for (const [label, re] of [["organization", /^syntx-(body-org|organization-)/], ["role", /^syntx-role-/], ["requirement", /^syntx-requirement-/]]) {
      const target = await tabTo(re); if (!target) { log.push({ label, target: null }); break; }
      await page.keyboard.press("Enter"); await page.waitForTimeout(1400); log.push({ label, target: target.own || target.id, after: await kind(), focus: await active() });
    }
    for (let i = 0; i < 3; i += 1) { await page.keyboard.press("Escape"); await page.waitForTimeout(1200); log.push({ label: `escape ${i + 1}`, after: await kind(), focus: await active() }); }
    res.descent.push({ w: W, log });
  });
}
writeFileSync(OUT, JSON.stringify(res, null, 1)); console.log("KEYS tab", res.tab.length, "descent", res.descent.length, "errors", res.errors.length); await browser.close();
