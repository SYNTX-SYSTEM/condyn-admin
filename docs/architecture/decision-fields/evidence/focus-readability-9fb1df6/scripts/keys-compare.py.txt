# PINK: pink-focus-keys.mjs on 38ebc31 vs 9fb1df6. python3 -I keys-compare.py <base keys.json> <tip keys.json>
import json,sys
b,a=[json.load(open(p)) for p in sys.argv[1:3]]
print('errors base',b['errors'],'tip',a['errors'])
seq=lambda t:[(s.get('own') or s['id'], s.get('aria','')[:40], s.get('ring'), s.get('inView'), s.get('kind')) for s in t['seq']]
for x,y in zip(b['tab'],a['tab']):
  sx,sy=seq(x),seq(y); same=sx==sy
  boxes=max((max(abs(p-q) for p,q in zip(u.get('box',[0]*4),v.get('box',[0]*4))) for u,v in zip(x['seq'],y['seq'])),default=0)
  print(f"tab {x['state']:12} {x['w']} start {x['start']['kind']}->{y['start']['kind']} stops {len(sx)} identical order/aria/ring/inView/kind: {same}; distinct {len(set(s[0] for s in sy))}; max focus-box shift {boxes} px")
  if not same:
    for i,(u,v) in enumerate(zip(sx,sy)):
      if u!=v: print('    first diff at',i,u,'|',v); break
  shifts=[(u.get('own') or u['id'],max(abs(p-q) for p,q in zip(u.get('box',[0]*4),v.get('box',[0]*4)))) for u,v in zip(x['seq'],y['seq'])]
  big=sorted({k:d for k,d in shifts if d>1}.items(),key=lambda kv:-kv[1])[:6]
  if big: print('    focus boxes moved >1 px:',big)
for x,y in zip(b['descent'],a['descent']):
  print('descent',x['w'])
  for u,v in zip(x['log'],y['log']):
    print(f"   {u['label']:12} target {u.get('target')}=={v.get('target')} kind {(u.get('after') or {}).get('kind')}->{(v.get('after') or {}).get('kind')} url-equal {(u.get('after') or {}).get('url')==(v.get('after') or {}).get('url')} focus {(u.get('focus') or {}).get('own')}->{(v.get('focus') or {}).get('own')}")
