# PINK: pink-focus-edges.mjs on 38ebc31 vs 121dc6d. python3 -I edges-compare.py <base edges.json> <tip edges.json>
import json,sys
b,a=[json.load(open(p)) for p in sys.argv[1:3]]
print('errors base',b['errors'],'\nerrors tip',a['errors'])
V={v['i']:v for v in a['variants']}
print('variants:'); [print(f"   {v['i']:2} pool {v['pool']['id'] if v['pool'] else '-':12} {v['pool']['len'] if v['pool'] else '':>3} {v['pool']['name'] if v['pool'] else ''!r:60} | org {v['org']['id'] if v['org'] else '-':12} label {v['org']['label'] if v['org'] else ''}: {v['org']['name'] if v['org'] else ''!r}") for v in a['variants']]
key=lambda r:(r['kind'],r['id'],r['w'],r['l'])
B={key(r):r for r in b['rows']}; A={key(r):r for r in a['rows']}
print('rows base',len(B),'tip',len(A),'missing in tip',sorted(set(B)-set(A)),'missing in base',sorted(set(A)-set(B)))
def show(kind,fields):
  print(f'\n== {kind}: base → tip')
  for k in sorted(k for k in A if k[0]==kind):
    x,y=B.get(k,{}),A[k]
    cols=' '.join(f"{f}={x.get(f)}→{y.get(f)}" if x.get(f)!=y.get(f) else f"{f}={y.get(f)}" for f in fields)
    col=f" COLL {len(x.get('collisions',[]))}→{len(y['collisions'])}" + (f" CLIP {len(x.get('clipped',[]))}→{len(y['clipped'])}" )
    print(f'   {k[1]:12} {k[2]:4} {k[3]} {cols}{col}')
for r in list(A.values())+list(B.values()):
  if r['kind']=='core' and 'ringAir' in r: r.update({'nameOuter':r['ringAir']['nameOuter'],'nameInner':r['ringAir']['nameInner'],'chipOuter':r['ringAir']['chipOuter'],'chipInner':r['ringAir']['chipInner']})
print('core: per-line air of the name against the organism body ring (220 px, nameOuter) and its inner membrane (184 px, nameInner), in screen px; negative = a line corner outside; overMax = widest line beyond the 176 px content width')
show('core',['font','lines','nameW','overMax','nameOuter','nameInner','chipOuter','chipInner','contentH','nameToChip','core'])
print('centre: per-line air against the 174 px disc (inside the 1.5 px border); overDisc = beyond the disc box')
show('centre',['font','lines','maxLineW','inner','overDisc','nameLineAir','kickerLineAir','nameBlock','kickerBlock','disc','nearestCard'])
print('\n== sibs')
for k in sorted(k for k in A if k[0]=='sibs'):
  x,y=B.get(k,{}),A[k]
  print(f"   {k[1]:14} {k[2]:4} {k[3]} obs {x.get('observation')}→{y['observation']} n {x.get('n')}→{y['n']} overlaps {x.get('overlaps')}→{y['overlaps']} onNucleus {x.get('onNucleus')}→{y['onNucleus']} COLL {len(x.get('collisions',[]))}→{len(y['collisions'])} CLIP {len(x.get('clipped',[]))}→{len(y['clipped'])}")
  for cx,cy in zip(x.get('cards',[]),y['cards']):
    print(f"      {cy['text'][:38]:38} font {cx['font']}→{cy['font']} lines {cx['lines']}→{cy['lines']} h {cx['h']}→{cy['h']} (min {cy['minH']}) w {cx['w']}→{cy['w']} textOut {cx['textOut']}→{cy['textOut']} off {cx['off']}→{cy['off']} op {cy['op']}")
print('\n== visible collisions / clipped on tip, listed')
for k in sorted(A):
  y=A[k]
  for c in y['collisions']: print('   COLL',k,c['a'][:70],'×',c['b'][:70],c['w'],c['h'], '(also base)' if any(d['a']==c['a'] and d['b']==c['b'] for d in B.get(k,{}).get('collisions',[])) else 'NEW')
  for c in y['clipped']: print('   CLIP',k,c, '(also base)' if c in B.get(k,{}).get('clipped',[]) else 'NEW')
