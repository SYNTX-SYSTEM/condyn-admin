# PINK: 38ebc31 vs 121dc6d from PINK's own frame measures of the walk (pink-typo-measure.mjs via pink-walk.mjs), one lab per viewport.
# python3 -I pink-focus-compare.py <out-base> <out-tip>
import json,sys,glob,os,re,collections
B,A=sys.argv[1:3]
MOVING=re.compile(r'^(syntx-role-|syntx-organization-|syntx-requirement-|syntx-trace|syntx-source|evidence-node|capability-node|capability-cosmos-node|syntx-open-|syntx-moon|subspace|semantic-node)')
load=lambda d:{os.path.basename(f):json.load(open(f)) for f in sorted(glob.glob(os.path.join(d,'*.m.json')))}
b,a=load(B),load(A); frames=sorted(set(b)&set(a))
print(f'frames base {len(b)} tip {len(a)} matched {len(frames)}; only base {sorted(set(b)-set(a))} only tip {sorted(set(a)-set(b))}')
clipped=lambda t:(t['clipPx']>0.5 and t.get('scrollPx',0)<=0.5 and t['clipBy']!='semantic-career-intelligence-field') or t['ellipsis'] or t['clamp']
tot=collections.Counter(); ch=collections.Counter(); dec=[]; newclip=[]; goneclip=[]; newcol=[]; gonecol=[]; moved=collections.defaultdict(float); unmatched=collections.Counter()
for fr in frames:
  tb={t['key']:t for t in b[fr]['texts']}; ta={t['key']:t for t in a[fr]['texts']}
  common=set(tb)&set(ta); tot['texts-base']+=len(tb); tot['texts-tip']+=len(ta); tot['matched']+=len(common)
  for k in set(tb)^set(ta): unmatched[re.sub(r'#\d+$','',k.split('|')[0])+(' (base only)' if k in tb else ' (tip only)')]+=1
  for k in common:
    x,y=tb[k],ta[k]
    if abs(x['fs']-y['fs'])>0.01:
      ch[(x['tp'].split('<')[0].rstrip('0123456789_abcdefghijklmnopqrstuvwxyzABCDEF') , x['text'][:44], x['fs'], y['fs'], len(x['lines']), len(y['lines']))]+=1
      if y['fs']<x['fs']: dec.append((fr,k))
    if clipped(y) and not clipped(x): newclip.append((fr,'(moving) ' if MOVING.match(x['tp']) else '',k[:100],y['clipPx'],y['clipBy'],y['ellipsis']))
    if clipped(x) and not clipped(y): goneclip.append((fr,k[:90]))
  cb={(c['a'],c['b']) for c in b[fr]['collisions'] if c['visible']}; ca={(c['a'],c['b']) for c in a[fr]['collisions'] if c['visible']}
  newcol+= [(fr,)+c for c in ca-cb]; gonecol+=[(fr,)+c for c in cb-ca]
  bb,ba=b[fr]['boxes'],a[fr]['boxes']
  if bb.get('semantic-career-intelligence-field',[0,0])[1]!=ba.get('semantic-career-intelligence-field',[0,0])[1]: tot['frames-shifted']+=1; continue
  for k in set(bb)&set(ba):
    if MOVING.match(k): continue
    d=max(abs(u-v) for u,v in zip(bb[k],ba[k])); kk=re.sub(r'#\d+$','',k); moved[kk]=max(moved[kk],d)
print('texts',dict(tot))
print('unmatched text keys (carrier: count):'); [print(f'   {n:4} {k}') for k,n in sorted(unmatched.items(),key=lambda kv:-kv[1])[:30]]
print('size changes (carrier | text | base→tip px | lines base→tip): frames'); [print(f'   {n:3}  {k[0]} | {k[1]} | {k[2]}→{k[3]} | {k[4]}→{k[5]}') for k,n in sorted(ch.items())]
print(f'DECREASED {len(dec)}', dec[:10])
print(f'new clipped {len(newclip)}'); [print('  ',*x) for x in newclip[:40]]
print(f'clipped no more {len(goneclip)}'); [print('  ',*x) for x in goneclip[:40]]
print(f'new visible collisions {len(newcol)}'); [print('  ',*x) for x in newcol[:40]]
print(f'visible collisions gone {len(gonecol)}'); [print('  ',*x) for x in gonecol[:40]]
print('non-moving testid boxes moved/resized > 1 px (max over frames):'); [print(f'   {d:7.1f} {k}') for k,d in sorted(moved.items(),key=lambda kv:-kv[1]) if d>1]
