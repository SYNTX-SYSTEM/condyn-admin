#!/bin/bash
# PINK full closure: tsc + whole isolated suite (incl. e2e) on 421e382 in PINK's worktree, disposable DBs, net sentinel.
S=${PINK_SCRATCH}; G=$S/fc/suite; P=${PINK_REPO}
cd $P; set -a; . ~/.config/condyn/test-db-role.env; set +a; unset GEMINI_API_KEY GEMINI_MODEL DATABASE_URL
for C in 421e382; do
  git checkout -q --detach $C && echo "HEAD $(git rev-parse --short HEAD)"
  npx tsc --noEmit -p . > $G/tsc-$C.log 2>&1; echo "tsc $C exit $? errors $(grep -c 'error TS' $G/tsc-$C.log): $(grep 'error TS' $G/tsc-$C.log | cut -c1-160 | tr '\n' ' ')"
  rm -f $G/net-$C.log
  env PINK_NET_LOG=$G/net-$C.log NODE_OPTIONS="--require $S/scope-paths/pink-net-sentinel.cjs" PINK_NET_SENTINEL=$S/scope-paths/pink-net-sentinel.cjs PINK_NET_CONTROL=$S/scope-paths/positive-control.mjs PINK_SCRIPTS_OUT=$G/scripts-$C npm run -s test:isolated -- --reporter=default --reporter=json --outputFile.json=$G/full-$C.json > $G/full-$C.log 2>&1; echo "full $C exit $? :: $(grep -E 'Test Files|Tests  ' $G/full-$C.log | tail -2 | tr '\n' ' ') net $(cat $G/net-$C.log 2>/dev/null | wc -l)"
  rm -f output2.txt; git status --short | head -5
done
git checkout -q validation/hr-decision-loop-pink && echo "back on $(git branch --show-current)"; echo SUITE-END
