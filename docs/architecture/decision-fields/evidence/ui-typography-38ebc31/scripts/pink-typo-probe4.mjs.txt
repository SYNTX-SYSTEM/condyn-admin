// PINK: low windows (about 150–250 % zoom) on the Job Field — the orientation surface's box, every painted text collision that involves
// a text inside the surface (PINK's measure), text cut inside the surface, and reachability of the edge actions without any scroll.
// argv: <lab state.json> <out.json>. Stand-in lab only.
import { readFileSync, writeFileSync } from "node:fs";
import { pathToFileURL } from "node:url";
import { measure } from "./pink-typo-measure.mjs";
const playwright = await import(pathToFileURL(process.env.CONDYN_PLAYWRIGHT_MODULE).href);
const { chromium } = playwright.chromium ? playwright : playwright.default;
const [STATE, OUT] = process.argv.slice(2);
const BASE = JSON.parse(readFileSync(STATE, "utf8")).base; const POOL = "JPOOL_B94850C439974E27496FE3B1A9A076B5";
await fetch(`${BASE}/api/career/job-pools`, { method: "POST", headers: { "content-type": "application/json", "x-condyn-principal-actor-id": "PINK_TYPO_LAB" }, body: readFileSync("docs/examples/job-pool-tests/DLL_JOBPOOL_ENGINEERING.json", "utf8") });
const browser = await chromium.launch(); const res = [];
const IDS = ["syntx-job-scope-select", "syntx-job-scope-clear", "syntx-pools-toggle", "job-pool-intake-toggle", "syntx-gemini-output-toggle", "syntx-investigation-toggle", "syntx-return-btn", "syntx-exit-btn"];
const STATES = [["role", `jobOrganizationId=org_kestrel&jobRoleId=role_kestrel_fullstack`], ["role-scoped-JOB", `jobOrganizationId=org_kestrel&jobRoleId=role_kestrel_frontend_architect&jobScopePoolUploadId=${POOL}&jobScopeRoleId=role_kestrel_frontend_architect&investigationMode=JOB`], ["organization", `jobOrganizationId=org_aurelion`], ["pool", ""]];
for (const [w, h] of [[768, 432], [960, 540], [1024, 576], [1152, 600], [1213, 600], [1214, 600], [1280, 600], [1280, 601], [1366, 600], [1536, 864]]) for (const lang of ["en", "de"]) for (const [st, q] of STATES) {
  const ctx = await browser.newContext({ viewport: { width: w, height: h } }); await ctx.addInitScript(() => { try { localStorage.setItem("condyn_onboarding_seen", "1"); } catch {} });
  const page = await ctx.newPage(); const t = id => page.getByTestId(id);
  try {
    await page.goto(`${BASE}/career/demo?focus=FIELD&jobPoolUploadId=${POOL}${q ? "&" + q : ""}`, { waitUntil: "networkidle" }); await t("field-focus-strip").waitFor({ timeout: 60000 });
    if (lang === "de") { await t("open-system-codex-btn").click(); await t("switch-lang-de").click(); await t("system-codex-close").click(); }
    await page.waitForTimeout(1200);
    const geo = await page.evaluate(ids => { const o = document.querySelector("[data-sx-orient]"); const ob = o.getBoundingClientRect(); const cs = getComputedStyle(o);
      const reach = {}; for (const id of ids) { const e = document.querySelector(`[data-testid="${id}"]`); if (!e || e.getBoundingClientRect().height === 0) continue; const q = e.getBoundingClientRect(); const x = (q.left + q.right) / 2, y = (q.top + q.bottom) / 2; const tp = y >= 0 && y < innerHeight && x < innerWidth ? document.elementFromPoint(x, y) : null; reach[id] = Boolean(tp && (tp === e || e.contains(tp))); }
      return { surface: [ob.left, ob.top, ob.right, ob.bottom].map(v => Math.round(v * 10) / 10), width: cs.width, overflowY: cs.overflowY, ownScroll: o.scrollHeight - o.clientHeight, reach }; }, IDS);
    const m = await measure(page);
    const inS = k => /syntx-path|syntx-job-scope|syntx-pools-toggle|job-pool-intake-toggle|syntx-gemini-output-toggle|syntx-investigation-toggle|syntx-return-btn|syntx-exit-btn|syntx-state|syntx-field\|/.test(k);
    const coll = m.collisions.filter(c => c.visible && (inS(c.a) || inS(c.b))).map(c => `${c.a.slice(0, 70)} × ${c.b.slice(0, 70)} ${c.w}x${c.h}`);
    const cut = m.texts.filter(x => inS(x.key) && ((x.clipPx > 0.5 && (x.scrollPx ?? 0) <= 0.5 && x.clipBy !== "semantic-career-intelligence-field") || x.ellipsis || x.clamp || x.offscreen > 0.5)).map(x => `${x.key.slice(0, 70)} clip ${x.clipPx} off ${x.offscreen}${x.ellipsis ? " ellipsis" : ""}`);
    res.push({ w, h, lang, st, ...geo, collisions: coll, cut });
  } catch (e) { res.push({ w, h, lang, st, error: String(e).split("\n")[0].slice(0, 200) }); }
  await ctx.close();
}
writeFileSync(OUT, JSON.stringify(res)); console.log(`PROBE4 ${res.length} errors ${res.filter(r => r.error).length}`); await browser.close();
