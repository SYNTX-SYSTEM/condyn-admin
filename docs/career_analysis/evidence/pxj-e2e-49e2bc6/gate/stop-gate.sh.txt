#!/usr/bin/env bash
# Stops the gate by the PID it recorded itself (never by pattern), then confirms the port is closed.
set -euo pipefail
pidfile="$1"
[ -f "$pidfile" ] || { echo "[stop-gate] no pid file"; exit 0; }
pid=$(python3 -I -c 'import json,sys;print(json.load(open(sys.argv[1]))["pid"])' "$pidfile")
port=$(python3 -I -c 'import json,sys;print(json.load(open(sys.argv[1]))["port"])' "$pidfile")
kill -TERM "$pid" 2>/dev/null || true
for _ in $(seq 1 30); do kill -0 "$pid" 2>/dev/null || break; sleep 0.1; done
if ss -ltnH "sport = :$port" | command grep -q .; then echo "[stop-gate] port $port still open"; exit 1; fi
rm -f "$pidfile"
echo "[stop-gate] stopped pid $pid, port $port closed"
