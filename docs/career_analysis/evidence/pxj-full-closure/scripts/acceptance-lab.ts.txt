/* OWNER ACCEPTANCE LAB (owner "SYNTX::SFE::OWNER_GO_FULL_PXJ_CLOSURE" §03, 2026-10-23). Run from the checkout under review:
 *
 *   set -a; . <test-db-role.env>; set +a; unset DATABASE_URL CONDYN_ALLOW_SHARED_DATABASE GEMINI_API_KEY GEMINI_MODEL GOOGLE_GEMINI_BASE_URL
 *   ACC_SOURCE_DB=<disposable lab database> ACC_PORT=<port> ACC_DIR=<dir> npx tsx <this file>
 *
 * The source must be a positively identified disposable test database (marker checked, read-only connection). It is copied with
 * pg_dump in a read-only session into a NEW disposable database; the source is never written. Every table is compared row for row
 * (count + md5 of the ordered rows) before the server starts. The server runs this checkout's code against the copy with no Gemini
 * key, no worker and the network sentinel: nothing can reach a model, ANALYZE cannot execute. SIGTERM stops the server and drops
 * only the copy. The shared condyn database is never touched. Writes <dir>/state.json (no secrets) and <dir>/copy-proof.json. */
import { spawn, spawnSync, type ChildProcess } from "node:child_process";
import { mkdirSync, writeFileSync, unlinkSync } from "node:fs";
import { resolve } from "node:path";

const W = process.cwd();
const SOURCE = process.env.ACC_SOURCE_DB ?? "";
const PORT = Number(process.env.ACC_PORT ?? "0");
const DIR = resolve(process.env.ACC_DIR ?? "");
const SENTINEL = resolve(W, "docs/career_analysis/evidence/video-readiness/scripts/net-sentinel.cjs");
if (!/^condyn_test_[0-9a-f]{16}$/.test(SOURCE) || !Number.isInteger(PORT) || PORT < 1024 || !process.env.ACC_DIR) { console.error("[acc-lab] ACC_SOURCE_DB (condyn_test_<16 hex>), ACC_PORT and ACC_DIR are required"); process.exit(2); }
const delay = (ms: number) => new Promise(r => setTimeout(r, ms));

const pgEnv = (url: string, readOnly: boolean) => {
  const u = new URL(url);
  return { ...process.env, PGHOST: u.hostname, PGPORT: u.port || "5432", PGUSER: decodeURIComponent(u.username), PGPASSWORD: decodeURIComponent(u.password), PGDATABASE: u.pathname.slice(1), ...(readOnly ? { PGOPTIONS: "-c default_transaction_read_only=on" } : {}) };
};

(async () => {
  mkdirSync(DIR, { recursive: true, mode: 0o700 });
  const { createDisposableTestDatabase, dropDisposableTestDatabase, verifyDisposableTestDatabase, disposableTestDatabaseUrl } = await import(`${W}/lib/database-isolation/verification`);
  const { requireTestDatabaseAdminUrl } = await import(`${W}/scripts/test-db/admin-url`);
  const basis = requireTestDatabaseAdminUrl();
  const source = await verifyDisposableTestDatabase(disposableTestDatabaseUrl(basis, SOURCE));
  const copy = await createDisposableTestDatabase(basis);
  const dropCopy = async () => { try { await dropDisposableTestDatabase(copy.url); } catch (error) { console.error("[acc-lab] drop failed", error instanceof Error ? error.message : error); } };
  try {
    const dump = spawnSync("pg_dump", ["--no-owner", "--no-privileges", "--format=plain"], { env: pgEnv(source.url, true), maxBuffer: 1 << 30 });
    if (dump.status !== 0) throw new Error(`pg_dump failed: ${String(dump.stderr).slice(0, 400)}`);
    const restore = spawnSync("psql", ["-X", "-q", "-v", "ON_ERROR_STOP=1", "--single-transaction"], { env: pgEnv(copy.url, false), input: dump.stdout, maxBuffer: 1 << 30 });
    if (restore.status !== 0) throw new Error(`restore failed: ${String(restore.stderr).slice(0, 400)}`);
    const fingerprint = (url: string) => {
      const q = `select string_agg(format('%s|%s', t, x), E'\\n' order by t) from (select c.relname t, (xpath('/row/x/text()', query_to_xml(format('select count(*)||'':''||coalesce(md5(string_agg(r::text, ''|'' order by r::text)),''-'') as x from %I.%I r', n.nspname, c.relname), false, true, '')))[1]::text x from pg_class c join pg_namespace n on n.oid=c.relnamespace where c.relkind='r' and n.nspname='public') s`;
      const out = spawnSync("psql", ["-X", "-At", "-c", q], { env: pgEnv(url, true), maxBuffer: 1 << 28 });
      if (out.status !== 0) throw new Error(`fingerprint failed: ${String(out.stderr).slice(0, 300)}`);
      return String(out.stdout).trim().split("\n");
    };
    const a = fingerprint(source.url), b = fingerprint(copy.url);
    const differing = a.filter((line, i) => line !== b[i]).concat(a.length === b.length ? [] : ["TABLE_COUNT"]);
    writeFileSync(resolve(DIR, "copy-proof.json"), JSON.stringify({ source: source.databaseName, copy: copy.databaseName, tables: a.length, identical: differing.length === 0, differing, tablesWithRows: a.filter(l => !/\|0:-$/.test(l)) }, null, 2));
    if (differing.length) throw new Error(`copy differs from source in ${differing.length} tables`);
  } catch (error) { console.error("[acc-lab]", error instanceof Error ? error.message : error); await dropCopy(); process.exit(1); }

  const env = { ...process.env, DATABASE_URL: copy.url, GEMINI_API_KEY: "", NEXT_TELEMETRY_DISABLED: "1", NODE_OPTIONS: `--require ${SENTINEL}`, LAB_NET_LOG: resolve(DIR, "net.jsonl"), TEST_DATABASE_ADMIN_URL: "", TEST_DATABASE_ROLE_PASSWORD: "" };
  // ACC_PROD=1: a local production build + `next start` of this checkout (measurement only, never a deployment).
  const PROD = process.env.ACC_PROD === "1";
  if (PROD) {
    const build = spawnSync(process.execPath, [resolve(W, "node_modules/next/dist/bin/next"), "build", "--webpack"], { cwd: W, env: { ...env, NODE_ENV: "production" }, stdio: ["ignore", "inherit", "inherit"] });
    if (build.status !== 0) { console.error(`[acc-lab] next build failed (${build.status})`); await dropCopy(); process.exit(1); }
  }
  const server: ChildProcess = spawn(process.execPath, [resolve(W, "node_modules/next/dist/bin/next"), PROD ? "start" : "dev", ...(PROD ? [] : ["--webpack"]), "-p", String(PORT), "-H", "127.0.0.1"], {
    cwd: W, stdio: ["ignore", "inherit", "inherit"], env: PROD ? { ...env, NODE_ENV: "production" } : env
  });
  let stopping = false;
  const stop = async () => {
    if (stopping) return; stopping = true;
    try { server.kill("SIGTERM"); } catch {}
    await delay(3000);
    if (server.exitCode === null) { try { server.kill("SIGKILL"); } catch {} }
    await dropCopy();
    try { unlinkSync(resolve(DIR, "state.json")); } catch {}
    process.exit(0);
  };
  process.on("SIGTERM", stop); process.on("SIGINT", stop);
  server.on("exit", () => { void stop(); });
  const base = `http://127.0.0.1:${PORT}`;
  for (let i = 0; i < 240; i += 1) { try { const r = await fetch(`${base}/api/career/job-pools`); if (r.status === 200) { await r.json(); break; } } catch {} await delay(500); }
  try { await (await fetch(`${base}/career/demo`)).text(); } catch {}
  const commit = String(spawnSync("git", ["rev-parse", "HEAD"], { cwd: W }).stdout).trim();
  writeFileSync(resolve(DIR, "state.json"), JSON.stringify({ port: PORT, base, commit, source: source.databaseName, database: copy.databaseName, pid: process.pid, serverPid: server.pid, gemini: "NONE", worker: "NONE", server: PROD ? "next start (local production build)" : "next dev" }, null, 2), { mode: 0o600 });
  console.log(`ACCEPTANCE LAB READY ${base} commit=${commit} copy=${copy.databaseName} source=${source.databaseName} gemini=NONE worker=NONE`);
})().catch(error => { console.error(error); process.exit(1); });
